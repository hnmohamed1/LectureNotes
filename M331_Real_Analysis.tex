	\subsection{Metric Space}
	\begin{definition}{Metric Space}By Metric Spaces $(M,d)$ we mean a non-empty set $M$ with function 
		\[
		d: M \times M \to \R
		\]
		such that:
		\begin{enumerate}
			\item \textbf{Non-negativity and Definiteness:}
			\[
			d(x,y)\geq 0, \quad \forall x,y \in M
			\]
			and
			\[
			d(x,y) = 0 \iff x=y
			\]
			
			\item \textbf{Symmetry:}
			\[
			d(x,y)=  d(y,x), \quad \forall x,y \in M
			\]
			
			\item \textbf{Triangle Inequality:}
			\[
			d(x,z)\leq d(x,y)+d(y,z), \quad \forall x,y,z \in M
			\]
		\end{enumerate}
	\end{definition}
	
	\begin{example}Let $M = \mathbb{R}$ and define the distance function as:
		\[ d(x, y) = |x - y| \]
		To show it is a metric, we verify the three main properties:
		\begin{enumerate}
			\item \textbf{Non-negativity:} 
			\[d(x, y) = |x - y| \geq 0\]
			and if \[d(x, y) = 0 \iff |x - y| = 0 \iff x = y.\]
			\item \textbf{Symmetry:}
			\[d(x, y) = |x - y| = |-(y - x)| = |y - x| = d(y, x)\]
			\item \textbf{Triangle Inequality:} 
			\[ d(x, z) = |x - z| = |(x - y) + (y - z)| \leq |x - y| + |y - z| = d(x, y) + d(y, z) \]
		\end{enumerate}
		% Muhammed Alaa El-Din: I made it explicit that the complex distance is the Euclidean metric d_2 on R^2 (identifying z = x + iy with (x, y)), whose triangle inequality is proved after the Cauchy--Schwarz inequality below.
		For any two complex numbers $z_1 = x_1 + iy_1$ and $z_2 = x_2 + iy_2$, the distance (Euclidean metric) is defined as:
		\[ |z_1 - z_2| = \sqrt{(x_1 - x_2)^2 + (y_1 - y_2)^2} \]
		Identifying $z = x + iy$ with $(x, y) \in \R^2$, this is exactly the Euclidean metric $d_2$ on $\R^2$ of the example below; its triangle inequality is proved in Corollary~\ref{cor:euclidtriangle}.
	\end{example}
	
	\begin{example}Let $M$ be any non-empty set and define the distance function as:
		\[ d(x, y) = \begin{cases} 0 & \text{if } x = y \\ 1 & \text{if } x \neq y \end{cases} \]
		
		This satisfies:
		\begin{enumerate}
			\item \textbf{Non-negativity:} it is clear that $d(x,y) \ge 0$. And, if $d(x,y) = 0$, then $x = y$ by the definition of the metric $d$.
			\item \textbf{Symmetry:} it is clear that $d(x,y) = d(y,x)$.
			\item \textbf{Triangle Inequality:} We want to show that $d(x, z) \leq d(x, y) + d(y, z)$ for any $x, y, z \in M$. We consider two cases:
			\begin{itemize}
				\item \textbf{Case 1 ($x = z$):} Then $d(x, z) = 0$. Since distances are non-negative ($d(x, y) \geq 0$ and $d(y, z) \geq 0$), their sum is $\geq 0$, so $0 \leq d(x, y) + d(y, z)$ holds.
				\item \textbf{Case 2 ($x \neq z$):} Then $d(x, z) = 1$. Since $x \neq z$, the point $y$ cannot be equal to both $x$ and $z$ simultaneously. Thus, at least one of $d(x, y)$ or $d(y, z)$ must equal $1$, meaning $d(x, y) + d(y, z) \geq 1$. Therefore, $1 \leq d(x, y) + d(y, z)$ holds.
			\end{itemize}
		\end{enumerate}
	\end{example}
	
	
	% Muhammed Alaa El-Din: I made a note on why these are metrics: non-negativity and symmetry are clear, but the triangle inequality for d_2 needs the Cauchy--Schwarz inequality (proved below) and for d_p it needs Minkowski's inequality; it fails for 0 < p < 1, which is why p >= 1 is required.
	\begin{example}On $\R^2$, for points $x = (x_1, x_2)$ and $y = (y_1, y_2)$, we may define different metrics:
		\begin{enumerate}
			\item \[ d_1(x, y) = |x_1 - y_1| + |x_2 - y_2| \]
			\item \[ d_2(x, y) = \sqrt{(x_1 - y_1)^2 + (x_2 - y_2)^2} \]
			\item \[ d_\infty(x, y) = \max(|x_1 - y_1|, |x_2 - y_2|) \]
			\item \[ d_p(x, y) = \left( |x_1 - y_1|^p + |x_2 - y_2|^p \right)^{1/p}, \quad p \ge 1 \]
		\end{enumerate}
		For $d_1$ and $d_\infty$ the triangle inequality follows directly from the one in $\R$. For $d_2$ it follows from the Cauchy--Schwarz inequality (Corollary~\ref{cor:euclidtriangle}), and for $d_p$ it is Minkowski's inequality (proof omitted). The condition $p \ge 1$ is essential: for $0 < p < 1$ the triangle inequality fails, e.g.\ for $p = \tfrac12$, $x = (0,0)$, $y = (1,0)$, $z = (1,1)$ we get $d_p(x,z) = 4 > 2 = d_p(x,y) + d_p(y,z)$.
	\end{example}
	
	
	\subsection{Cauchy--Schwarz Inequality}
	\begin{theorem}[Cauchy--Schwarz Inequality]
		For any  $(a_1, a_2, \dots, a_n)$ and $(b_1, b_2, \dots, b_n)$ in $\mathbb{R}^n$, the Cauchy--Schwarz inequality states:
		\[ \left( \sum_{i=1}^{n} a_i b_i \right)^2 \leq \left( \sum_{i=1}^{n} a_i^2 \right) \left( \sum_{i=1}^{n} b_i^2 \right) \]
	\end{theorem}
	
	% Muhammed Alaa El-Din: I made the case C = 0 come first, so that the discriminant argument is only used when it is valid (it needs a genuine quadratic, i.e. C > 0).
	\begin{proof}
		Let $A = \sum_{i=1}^{n} a_i^2$, $B = \sum_{i=1}^{n} a_i b_i$, and $C = \sum_{i=1}^{n} b_i^2$.
		
		\noindent If $C = 0$, then all $b_i = 0$, and the inequality holds trivially since both sides equal zero. So assume $C > 0$.
		
		\noindent Consider the non-negative quadratic expression for any real parameter $t$:
		\[ \sum_{i=1}^{n} (a_i - t b_i)^2 \geq 0 \]
		
		\noindent Expanding this gives:
		\[ \sum_{i=1}^{n} \left(a_i^2 - 2t a_i b_i + t^2 b_i^2\right) = \left(\sum_{i=1}^{n} a_i^2\right) - 2t \left(\sum_{i=1}^{n} a_i b_i\right) + t^2 \left(\sum_{i=1}^{n} b_i^2\right) \geq 0 \]
		
		\noindent that is, the quadratic in $t$
		\[ Ct^2 - 2Bt + A \geq 0 \qquad \forall t \in \R. \]
		
		\noindent Since $C > 0$, for this quadratic to remain non-negative for all real $t$, its discriminant $\Delta = (-2B)^2 - 4AC$ must be less than or equal to zero:
		\[ 4B^2 - 4AC \leq 0 \implies B^2 \leq AC \]
		
		\noindent Substituting back $A$, $B$, and $C$ yields the Cauchy--Schwarz inequality:
		\[ \left( \sum_{i=1}^{n} a_i b_i \right)^2 \leq \left( \sum_{i=1}^{n} a_i^2 \right) \left( \sum_{i=1}^{n} b_i^2 \right). \qedhere \]
	\end{proof}
	
	% Muhammed Alaa El-Din: I made this new corollary: it is the reason the Cauchy--Schwarz inequality appears in this lecture, and without it the claim that d_2 (and the complex distance) is a metric was never proved.
	\begin{corollary}[Triangle Inequality for the Euclidean Metric]
		\label{cor:euclidtriangle}
		For $x, y \in \R^n$ put $d_2(x,y) = \bigl(\sum_{i=1}^n (x_i - y_i)^2\bigr)^{1/2}$. Then $d_2$ is a metric on $\R^n$.
	\end{corollary}
	
	\begin{proof}
		Non-negativity, definiteness and symmetry are clear. For the triangle inequality, put $a_i = x_i - y_i$ and $b_i = y_i - z_i$, so $a_i + b_i = x_i - z_i$. By Cauchy--Schwarz,
		\[
		\sum_{i=1}^n (a_i + b_i)^2 = \sum a_i^2 + 2\sum a_i b_i + \sum b_i^2 \le \sum a_i^2 + 2\Bigl(\sum a_i^2\Bigr)^{1/2}\Bigl(\sum b_i^2\Bigr)^{1/2} + \sum b_i^2 = \Bigl[\Bigl(\sum a_i^2\Bigr)^{1/2} + \Bigl(\sum b_i^2\Bigr)^{1/2}\Bigr]^2 .
		\]
		Taking square roots gives $d_2(x,z) \le d_2(x,y) + d_2(y,z)$.
	\end{proof}

	\subsection{Open Balls and Bounded Sets}
	
	\begin{definition}{Open Ball}Let $(M,d)$ be a metric space, $x_0 \in M$ and $r \in \mathbb{R}^{+}$. The set defined by 
		\[
		B(x_0; r) := \{x \in M : d(x_0, x) < r\}
		\]
		is called the open ball centered at $x_0$ with radius $r$.
	\end{definition}
	
	\begin{remark}Boundedness in $\mathbb{R}$:
		A subset $E \subseteq \mathbb{R}$ is bounded if and only if any of the following equivalent conditions hold:
		\begin{enumerate}
			\item There exist real numbers $l, L \in \mathbb{R}$ such that:
			\[
			l < x < L \quad \forall x \in E
			\]
			\item There exists a positive real number $K > 0$ such that:
			\[
			-K < x < K \quad (\text{or } |x| < K) \quad \forall x \in E
			\]
		\end{enumerate}
	\end{remark}
	
	\begin{definition}{Bounded Set}A subset $E$ of a metric space $(M,d)$ is called bounded if it is contained in a ball $E \subseteq B(x_0; r)$ for some $x_0 \in M$ and $r \in \mathbb{R}^{+}$.
	\end{definition}
	
	% Muhammed Alaa El-Din: I made the finite case explicit (by induction), since the statement mentions a finite number of sets but the proof only treats two; this finite version is used in Lecture 7.
	\begin{theorem}
		\label{thm:unionbounded}
		The union of any two (finite number of) bounded set is bounded
	\end{theorem}
	\begin{proof}
		Let $E_1$ and $E_2$ be bounded sets, that is, $\exists x_1, x_2 \in M;~ r_1,r_2\in\R^+$ with
		\[
		E_k \subseteq B(x_k;r_k), \quad k=1,2
		\]
		
		\noindent It suffices to find one open ball containing both of these balls. Let:
		\[
		R = \max \{ d(x_1, x_2) + r_2 , r_1 \}
		\]
		Then $B(x_1; r_1) \cup B(x_2; r_2) \subseteq B(x_1; R)$. \medskip
		
		\noindent For if $x \in B(x_1; r_1)$, then $d(x_1, x) < r_1 \le R$. If $x \in B(x_2; r_2)$, then $d(x_2, x) < r_2$, and by the triangle inequality
		\[
		d(x_1, x) \le d(x_1, x_2) + d(x_2, x) < d(x_1, x_2) + r_2 \le R.
		\]
		
		\noindent So in both cases $x \in B(x_1; R)$. Hence $E_1 \cup E_2 \subseteq B(x_1; R)$, and the union is bounded.
		
		\medskip
		\noindent For finitely many bounded sets $E_1, \dots, E_m$, induction on $m$ applies: $E_1 \cup \cdots \cup E_m = (E_1 \cup \cdots \cup E_{m-1}) \cup E_m$ is the union of two bounded sets.
	\end{proof}
	
	\subsection{Interior Points and Open Sets}
	
	\begin{definition}{Interior Point}Let $E \subseteq M$ where $(M,d)$ is a metric space. A point $x_0 \in E$ is called an interior point of $E$ if there is $\varepsilon > 0$ such that $B(x_0; \varepsilon) \subseteq E$. The set of all interior points of $E$ is denoted by $E^\circ$.
	\end{definition}
	
	\begin{definition}{Open Set}A subset $G$ of a metric space $(M,d)$ is called an open set if all its points are interior points, that is, $G = G^\circ$.
	\end{definition}
	
	
	\begin{definition}{Intervals}$I \subseteq \R $ is called an interval if 
		\[
		\forall x,y,z \in  \R, \quad x < y < z, \quad x,z \in I \implies y \in I

	\begin{remark}Basic relationships for interior sets:

		\]
	\end{remark}
	
	\begin{theorem}[Properties of Open Sets]
		\begin{enumerate}
			\item $\emptyset$ and $M$ are open sets.
			\item The intersection of any two (finite number of) open sets is open.
			\item The arbitrary union of open sets is open.
		\end{enumerate}
	\end{theorem}
	
	\begin{proof}
		\begin{enumerate}
			\item \textbf{For $\emptyset$ and $M$:} 
			The empty set $\emptyset$ contains no points, so the condition of containing an open ball around every point is vacuously true. For the space $M$ itself, given any point $x \in M$ and any radius $r > 0$, the open ball $B(x; r)$ is a subset of $M$ by definition ($B(x; r) \subseteq M$). Thus, every point in $M$ is an interior point, making $M$ an open set.
			
			\item \textbf{For Finite Intersection:} 
			Let $G_1$ and $G_2$ be two open sets and let $x_0 \in G_1 \cap G_2$. Then $x_0 \in G_1$ and $x_0 \in G_2$. \medskip
			
			\noindent Since $G_1$ and $G_2$ are open, there exist $r_1, r_2 > 0$ such that:
			\[
			B(x_0; r_1) \subseteq G_1 \quad \text{and} \quad B(x_0; r_2) \subseteq G_2
			\]
			Taking $r = \min(r_1, r_2)$, we obtain:
			\[
			B(x_0; r) = B(x_0; r_1) \cap B(x_0; r_2) \subseteq G_1 \cap G_2
			\]
			Therefore, $x_0$ is an interior point of $G_1 \cap G_2$, which proves that $G_1 \cap G_2$ is open.
			% Muhammed Alaa El-Din: I made the extension to finitely many sets explicit, and added why the word "finite" matters.
			For finitely many open sets $G_1, \dots, G_m$ the same argument works with $r = \min\{r_1, \dots, r_m\} > 0$. (For infinitely many sets the infimum of the radii may be $0$: e.g.\ $\bigcap_{n=1}^\infty \left]-\tfrac1n, \tfrac1n\right[ = \{0\}$ is not open in $\R$.)

			\vspace{0.5em}
			
			\item \textbf{For Arbitrary Union:} 
			Let $\{G_i\}_{i \in I}$ be a collection of open sets and let $G = \bigcup_{i \in I} G_i$. If $x \in G$, then there exists an index $k \in I$ such that $x \in G_k$. \medskip
			
			\noindent Since $G_k$ is open, there exists an open ball around $x$ that is contained in $G_k$, and since $G_k \subseteq G$, this ball is also contained in $G$. Thus, every point in $G$ is an interior point, proving that $G$ is open.
		\end{enumerate}
	\end{proof}

	\subsection{Continue: Properties of Open Sets}
	
	\begin{takeaway}
		Let $(M,d)$ be a metric space. Then:
		\begin{enumerate}[label=(\arabic*)]
			\item $\emptyset$ and $M$ are open.
			\item A finite intersection of open sets is open.
			\item An arbitrary union of open sets is open.
		\end{enumerate}
	\end{takeaway}
	
	\begin{theorem}
		\label{thm:openballopenset}
		Any open ball is an open set.
	\end{theorem}
	
	\begin{proof}
		Let $G = B(x_0;R)$ and let $x \in G$. Put
		\[
		r = \tfrac{1}{2}\bigl[R - d(x,x_0)\bigr] > 0 .
		\]
		If $y \in B(x;r)$, then by the triangle inequality
		\[
		d(y,x_0) \le d(y,x) + d(x,x_0) < r + d(x,x_0) < R,
		\]
		so $y \in B(x_0;R)$. Hence $B(x;r) \subseteq G$, so $x$ is an interior point of $G$.
		Since $x \in G$ was arbitrary, $G$ is open.		
	\end{proof}
	
	\noindent Thus, the union of any family of open balls is an open set.
	
	\begin{theorem}
		Any open set can be written as a union of open balls.
	\end{theorem}
	
	\begin{proof}
		Let $G$ be an open set. Then for every $x \in G$ there is $r_x \in \R^+$ such that $B(x;r_x) \subseteq G$. Now,
		\[
		G = \bigcup_{x\in G}\{x\} \subseteq \bigcup_{x\in G} B(x;r_x) \subseteq G .
		\]
		Therefore
		\[
		G = \bigcup_{x\in G} B(x;r_x)
		\]
	\end{proof}

	\begin{definition}{Closed Set}
		A set is called \textbf{closed} if its complement is open.
	\end{definition}
	
	\begin{takeaway}

			\item An arbitrary intersection of closed sets is closed.

	% Muhammed Alaa El-Din: I made a short justification of these properties (they were stated without proof): they follow from the properties of open sets by De Morgan's laws.
	\begin{proof}[Justification]
		Take complements and use De Morgan's laws:
		\[
		\Bigl(\bigcup_{k=1}^{m} F_k\Bigr)^c = \bigcap_{k=1}^{m} F_k^c, \qquad \Bigl(\bigcap_{i\in I} F_i\Bigr)^c = \bigcup_{i\in I} F_i^c .
		\]
		If the $F$'s are closed, the right-hand sides are a finite intersection, respectively an arbitrary union, of open sets, hence open. Finally $M^c = \emptyset$ and $\emptyset^c = M$ are open.
	\end{proof}

	\subsection{Accumulation Points of a Set}
	
	\begin{definition}{Accumulation Point}
		A point $x_0 \in M$ is called an \textbf{accumulation point} of a set $E \subseteq M$ if every
		open ball centered at $x_0$ contains a point of $E$ different from $x_0$, that is,
		\[
		\forall r>0 \ \ \exists\, x_1 \in E,\ x_1 \neq x_0,\ \text{ such that } x_1 \in B(x_0;r).
		\]
		Equivalently, $B^*(x_0;r)\cap E \neq \emptyset$ for every $r>0$, where
		$B^*(x_0;r)=B(x_0;r)\setminus\{x_0\}$ is the deleted ball.
	\end{definition}
	
	\begin{example}
		Consider $\R$ (with the usual metric $d(x,y)=|x-y|$). Let
		\[
		E = \Bigl\{1,\tfrac12,\tfrac13,\dots,\tfrac1n,\dots\Bigr\} = \Bigl\{\tfrac1n : n\in\Z^+\Bigr\}.
		\]
		\begin{enumerate}[label=(\arabic*)]
			\item \textbf{$x_0 = 0$ is an accumulation point of $E$.} Let $r\in\R^+$. By the Archimedean
			property there is $n_0\in\Z^+$ with $n_0 > \tfrac1r$, so
			\[
			0 < \tfrac{1}{n_0} < r, \qquad\text{i.e.}\qquad \tfrac{1}{n_0}\in B^*(0;r)\cap E .
			\]
			\item \textbf{This set has no other accumulation points.}
			\begin{enumerate}[label=(\roman*)]
				\item Let $x_0<0$. Take $r=\tfrac12|x_0|$. Then $B(x_0;r)$ contains no point from $E$.
				\item Let $x_0>1$. Take $r=\tfrac12(x_0-1)$. Then $B(x_0;r)\cap E=\emptyset$.
				% Muhammed Alaa El-Din: I made the choice of r justified: the neighbour of 1/n_0 on the other side, 1/(n_0 - 1), is even farther away, so only the gap to 1/(n_0 + 1) matters.
				\item Let $x_0\in E$. Then $x_0=\tfrac{1}{n_0}$ for some $n_0\in\Z^+$. Take
				\[
				r=\tfrac12\Bigl(\tfrac1{n_0}-\tfrac1{n_0+1}\Bigr)=\frac{1}{2n_0(n_0+1)} .
				\]
				Then $B^*(x_0;r)\cap E=\emptyset$, because the nearest points of $E$ to $x_0$ are $\tfrac1{n_0+1}$ (at distance $\tfrac{1}{n_0(n_0+1)} > r$) and, if $n_0 \ge 2$, $\tfrac1{n_0-1}$ (at distance $\tfrac{1}{n_0(n_0-1)} > r$).
				\item If $x_0\notin E$ lies between $\tfrac1{n_0+1}$ and $\tfrac1{n_0}$, take
				\[
				r=\tfrac12\min\Bigl\{\tfrac1{n_0}-x_0,\; x_0-\tfrac1{n_0+1}\Bigr\}.
				\]
				Then $B(x_0;r)\cap E=\emptyset$.
			\end{enumerate}
			In each case $x_0$ is not an accumulation point of $E$. (Every $x_0 \neq 0$ falls into one of the cases (i)--(iv).)
		\end{enumerate}
	\end{example}

	\subsection{Interior and Closure}
	
	\begin{theorem}
		The interior of any set $A$ in a metric space is the largest open set contained in $A$. That is,
		\[
		A^\circ = \bigcup_{\substack{G\subseteq A\\ G\ \text{is open}}} G .
		\]
	\end{theorem}
	
	% Muhammed Alaa El-Din: I made two gaps explicit: the monotonicity G_0 \subseteq A => G_0^\circ \subseteq A^\circ is now justified, and I added the final step showing that the formula really means "A^\circ is the largest open set in A" (in particular, A^\circ is open).
	\begin{proof}
		\textbf{(i)} Let $x_0\in A^\circ$. Then there is an open ball $B(x_0;r)$ such that $B(x_0;r)\subseteq A$. Since open balls are open sets (Theorem~\ref{thm:openballopenset}), $B(x_0;r)$ is an open set contained in $A$, so $x_0$ belongs to the right-hand side. Hence
		\[
		A^\circ \subseteq \bigcup_{\substack{G\subseteq A\\ G\ \text{is open}}} G .
		\]
		
		\textbf{(ii)} Let $x_0\in\bigcup_{G\subseteq A,\ G\text{ open}} G$. Then there exists an open set $G_0\subseteq A$ such that $x_0\in G_0$.
		
		Note that $G_0\subseteq A \Rightarrow G_0^\circ\subseteq A^\circ$ (if $B(x;r)\subseteq G_0$, then $B(x;r)\subseteq A$). But $G_0^\circ = G_0$ (since $G_0$ is open). Hence $x_0\in G_0\subseteq A^\circ$.
		
		\medskip
		\noindent \textbf{(iii)} By (i) and (ii), $A^\circ$ is a union of open sets, hence open, and it is contained in $A$; and every open $G\subseteq A$ appears in the union, so $G\subseteq A^\circ$. Thus $A^\circ$ is the largest open set contained in $A$.
	\end{proof}
	
	
	\begin{definition}{Closure}
		The smallest closed set containing $E$ is called the \textbf{closure} of $E$, denoted $\cl{E}$:
		\[
		\cl{E} = \bigcap_{\substack{F\supseteq E\\ F\ \text{is closed}}} F .
		\]
		(This intersection is closed, since an arbitrary intersection of closed sets is closed.)
	\end{definition}
	
	\begin{theorem}
		\label{thm:unionofsetanditsacc.}
		For every set $E$ in a metric space, $\cl{E} = E\cup E'$.
	\end{theorem}

	\subsection{Sequences and Convergence}
	
	\begin{definition}{Sequence}
		A \textbf{sequence} in a metric space $(M,d)$ is a function from $\Z^+$ to $M$. The terms of such a function can be written as
		\[
		x_1,\, x_2,\, x_3,\, \dots \qquad\text{instead of}\qquad x(1),\, x(2),\, x(3),\, \dots
		\]
		Sometimes we write the whole sequence as $\{x_n\}_{n\in\Z^+}$ or $(x_n)_{n=1}^{\infty}$.
	\end{definition}
	
	\begin{caution}
		The notation $\{x_n : n\in\Z^+\}$ (a set of values) is \textbf{not} a sequence: a set forgets order and
		repetitions. By $\{x_n\}_{n\in\Z^+}$ we always mean the ordered family $(x_n)_{n=1}^\infty$.
	\end{caution}
	\begin{definition}{Convergent Sequence}
		A sequence $\{x_n\}_{n\in\Z^+}$ is called \textbf{convergent} in the space $(M,d)$ if there is an $x_0\in M$ such that
		\[
		\forall\, \varepsilon>0 \ \ \exists\, N_\varepsilon\in\Z^+ \ \text{ s.t. } \ \forall\, n\ge N_\varepsilon \quad d(x_n,x_0)<\varepsilon .
		\]
		In this case we write
		\[
		\lim_{n\to\infty} x_n = x_0 .
		\]
	\end{definition}
	
	\begin{theorem}
		The limit is unique (if it exists).
	\end{theorem}
	
	\begin{proof}
		Suppose that $\lim x_n = x_0$ and $\lim x_n = x_0'$. Then for any $\varepsilon>0$ there exist $N_\varepsilon', N_\varepsilon''\in\Z^+$ such that
		\begin{align*}
			\forall n\ge N_\varepsilon' &\quad d(x_n,x_0) < \varepsilon/2, \\
			\forall n\ge N_\varepsilon'' &\quad d(x_n,x_0') < \varepsilon/2 .
		\end{align*}
		Let $N_\varepsilon=\max\{N_\varepsilon',N_\varepsilon''\}$. Then
		\[
		0 \le d(x_0,x_0') \le d(x_0,x_{N_\varepsilon}) + d(x_{N_\varepsilon},x_0') < \frac{\varepsilon}{2}+\frac{\varepsilon}{2} = \varepsilon .
		\]
		Since $\varepsilon>0$ is arbitrary, $d(x_0,x_0')=0$, hence $x_0=x_0'$.
	\end{proof}
	
	\begin{definition}{Divergent Sequence}
		A sequence which is not convergent is called \textbf{divergent}. That is,
		\[
		\forall\, x_0\in M \ \ \exists\, \varepsilon>0 \ \ \forall\, N\in\Z^+ \ \ \exists\, n\ge N \ \text{ s.t. } \ d(x_0,x_n)\ge\varepsilon .
		\]
	\end{definition}
	
	\begin{remark}
		Convergence and divergence depend on the set $M$ and the distance $d$.
	\end{remark}
	
	\begin{example}[Discrete Metric]
		Let $(M,d)$ be a discrete metric space, i.e.\ $d(x,y)=0$ if $x=y$ and $d(x,y)=1$ if $x\neq y$. Then a sequence is convergent if and only if it is eventually constant.
		
		($\Rightarrow$) Suppose $\lim x_n=x_0$ and take $\varepsilon=\tfrac12$. There is $N$ such that
		$d(x_n,x_0)<\tfrac12$ for all $n\ge N$. Since $d$ only takes the values $0$ and $1$,
		this forces $d(x_n,x_0)=0$, i.e.\ $x_n=x_0$ for all $n\ge N$. So $(x_n)$ is eventually constant.
		
		($\Leftarrow$) If $x_n=x_0$ for all $n\ge N$, then $d(x_n,x_0)=0<\varepsilon$ for every $\varepsilon>0$ and all $n\ge N$, so $\lim x_n=x_0$.
	\end{example}
	
	% =====================================================
	\subsection{Cauchy Sequences and Completeness}
	
	\begin{definition}{Cauchy Sequence}
		Let $\{x_n\}_{n\in\Z^+}$ be a sequence in a metric space $(M,d)$. It is called \textbf{Cauchy} if for every $\varepsilon>0$ there is an $N_\varepsilon\in\Z^+$ such that
		\[
		\forall\, m,n\ge N_\varepsilon \quad d(x_n,x_m)<\varepsilon .
		\]
	\end{definition}
	
	\begin{theorem}
		Any convergent sequence is Cauchy.
	\end{theorem}
	
	\begin{proof}
		Suppose that $\lim x_n=x_0$. Then for every $\varepsilon>0$ there is $N_\varepsilon\in\Z^+$ such that
		\begin{align*}
			\forall n\ge N_\varepsilon &\quad d(x_n,x_0)<\varepsilon/2,\\
			\forall m\ge N_\varepsilon &\quad d(x_m,x_0)<\varepsilon/2 .
		\end{align*}
		Hence for $m,n\ge N_\varepsilon$,
		\[
		d(x_m,x_n)\le d(x_m,x_0)+d(x_0,x_n)<\frac{\varepsilon}{2}+\frac{\varepsilon}{2}=\varepsilon .
		\]
	\end{proof}
	
	\begin{definition}{Complete Metric Space}
		A metric space $(M,d)$ is called \textbf{complete} if every Cauchy sequence in $M$ is convergent.
	\end{definition}
	
	% Muhammed Alaa El-Din: I made explicit that the Cauchy property in Q is inherited from R (both use the same metric), which is the real reason the sequence is Cauchy in Q.
	\begin{example}
		$\R$ with the usual metric is complete, but $\Q$ with the usual metric is not. Consider the sequence of rational numbers $x_n=\bigl(1+\tfrac1n\bigr)^n$. It converges in $\R$ (to the irrational number $e$), so it is Cauchy in $\R$, and hence also Cauchy in $\Q$ (the Cauchy condition only involves the distances $|x_n - x_m|$, which are the same in $\Q$ and in $\R$). But it has no limit in $\Q$: a limit $q\in\Q$ would also be a limit in $\R$, and limits are unique, so $q = e\notin\Q$.
	\end{example}
	
	% =====================================================
	\subsection{Closure and Completeness of Subsets}
	
	\begin{theorem}
		\label{thm:closure}
		Let $(M,d)$ be a metric space and $A\subseteq M$. Then $x_0\in \cl{A}$ if and only if there is a sequence $\{x_n\}_{n\in\Z^+}\subseteq A$ such that $\lim x_n=x_0$.
	\end{theorem}

	\end{exercise}
	
	\begin{corollary}
		\label{cor:completemetricandclosness}
		A subset of a complete metric space is complete if and only if it is closed.
	\end{corollary}

		Alternatively, one can argue directly: a Cauchy sequence in $\Z$ is eventually constant (take $\varepsilon = \tfrac12$; two integers at distance $<\tfrac12$ are equal), hence convergent.

	\subsection{Subsequences}
	
	\begin{definition}{Subsequence}
		Let $\{k_n\}_{n=1}^{\infty}$ be a strictly increasing sequence of positive integers, i.e.
		\[
		k_1 < k_2 < k_3 < \cdots < k_n < \cdots
		\]
		Given a sequence $\{x_n\}_{n=1}^{\infty}$ and a strictly increasing sequence of indices $\{k_n\}_{n=1}^{\infty}$, we say that $\{x_{k_n}\}_{n=1}^{\infty}$ is a \textbf{subsequence} of $\{x_n\}_{n=1}^{\infty}$.
	\end{definition}
	
	\begin{example}
		The sequences $\{x_{2n}\}$, $\{x_{2n+1}\}$, $\{x_{n^2}\}$ and $\{x_{n^3}\}$ are subsequences of $\{x_n\}$.
	\end{example}
	
	\begin{definition}{Limit Point of a Sequence}
		In a metric space $(M,d)$, a point $x_0$ is called a \textbf{limit point} of the sequence $\{x_n\}_{n=1}^{\infty}\subseteq M$ if this sequence has a subsequence $\{x_{k_n}\}$ that tends to $x_0$.
	\end{definition}
	
	% Muhammed Alaa El-Din: I made the claim "two limit points" complete: the original only showed that 1 and -1 are limit points, not that there are no others.
	\begin{example}
		The sequence $1,-1,1,-1,\dots$, that is $\{(-1)^n\}_{n=1}^{\infty}$, has two limit points, $1$ and $-1$, since
		\[
		x_{2n}=1, \qquad x_{2n+1}=-1 \qquad \forall n\in\Z^+ .
		\]
		There are no others: every term lies in $\{1,-1\}$, so for $x_0\notin\{1,-1\}$ and $r = \min\{|x_0-1|,|x_0+1|\}>0$, no term lies in $B(x_0;r)$, and no subsequence can tend to $x_0$.
	\end{example}
	
	% Muhammed Alaa El-Din: I made the caution concrete with an example showing the two notions really differ.
	\begin{caution}
		A \emph{limit point of a sequence} (a limit of some subsequence) is not the same notion as an \emph{accumulation point of a set} (Lecture 3). For example, the constant sequence $x_n = 5$ has the limit point $5$, but its set of values $\{5\}$ has no accumulation points.
	\end{caution}
	
	\subsection{Subsequences and Convergence}
	
	\begin{lemma}
		If $\{k_n\}$ is a strictly increasing sequence of positive integers, then $k_n\ge n$ for all $n\in\Z^+$.
	\end{lemma}
	
	\begin{proof}
		We use induction on $n$. Since $k_1$ is a positive integer, $k_1\ge 1$. Suppose $k_m\ge m$ for some $m$. Then
		\[
		k_{m+1} > k_m \ge m \implies k_{m+1} > m \implies k_{m+1}\ge m+1 ,
		\]
		because the $k_n$ are integers.
	\end{proof}
	
	\begin{theorem}
		\label{thm:subseq}
		The sequence $\{x_n\}_{n=1}^{\infty}$ converges to $x_0$ if and only if every subsequence of $\{x_n\}$ converges to $x_0$.
	\end{theorem}
	
	\begin{proof}
		\textbf{($\Rightarrow$)} Suppose that $x_n\to x_0$. Thus
		\[
		\forall \varepsilon>0 \ \ \exists N_\varepsilon\in\Z^+ \ \text{ s.t. } \ \forall n\in\Z^+ \quad n\ge N_\varepsilon \implies d(x_n,x_0)<\varepsilon .
		\]
		Let $\{x_{k_n}\}$ be any subsequence. Since $k_n\ge n$, we have
		\[
		\forall n\in\Z^+ \quad n\ge N_\varepsilon \implies k_n\ge N_\varepsilon \implies d(x_{k_n},x_0)<\varepsilon ,
		\]
		so $x_{k_n}\to x_0$.
		
		\medskip
		\noindent\textbf{($\Leftarrow$)} Suppose that all subsequences of $\{x_n\}_{n=1}^{\infty}$ tend to $x_0$. Then $\{x_n\}$, being a subsequence of itself (take $k_n=n$), tends to $x_0$.
	\end{proof}
	
	\begin{theorem}
		\label{thm:cauchylimitpoint}
		A Cauchy sequence cannot have more than one limit point.
	\end{theorem}

		Prove Theorem~\ref{thm:cauchylimitpoint}. \textit{(Assignment.)}
	\end{exercise}
	
	\begin{theorem}
		\label{thm:cauchy_converges}
		If a Cauchy sequence has a limit point, then it converges.
	\end{theorem}
	
	\begin{exercise}
		Prove Theorem~\ref{thm:cauchy_converges}. \textit{(Assignment.)}
	\end{exercise}

	\subsection{Equivalent Metrics}
	
	\begin{definition}{Equivalent Metrics}
		Two metrics $d,\rho$ on the same non-empty set $M$ are called \textbf{equivalent} if there exist $\alpha,\beta\in\R^+$ such that
		\[
		\alpha\, d(x,y)\le \rho(x,y)\le \beta\, d(x,y) \qquad \forall x,y\in M .
		\]
	\end{definition}
	
	% Muhammed Alaa El-Din: I made a remark on terminology: many textbooks call this notion "strongly" (or "Lipschitz") equivalent and reserve "equivalent" for metrics with the same open sets; the notion here implies that one.
	\begin{remark}
		Many textbooks call this notion \emph{strongly equivalent} (or \emph{Lipschitz equivalent}) and use the word ``equivalent'' for the weaker condition that $d$ and $\rho$ have the same open sets. Metrics that are equivalent in the sense above always have the same open sets and the same convergent sequences (see the exercises below), but the converse is false.
	\end{remark}
	
	\begin{example}
		On $M=\R^2$, the two metrics $d_1$ and $d_\infty$ are equivalent, where
		\begin{align*}
			d_1\bigl((x_1,y_1),(x_2,y_2)\bigr) &= |x_1-x_2|+|y_1-y_2| ,\\
			d_\infty\bigl((x_1,y_1),(x_2,y_2)\bigr) &= \max\{|x_1-x_2|,\,|y_1-y_2|\} .
		\end{align*}
		Indeed,
		\[
		d_\infty \le d_1 \le 2\, d_\infty .
		\]
	\end{example}
	
	\begin{exercise}
		What about $d_2$ (the Euclidean metric)? Is it equivalent to $d_1$ and $d_\infty$?
	\end{exercise}
	
	\begin{exercise}
		\begin{enumerate}[label=(\arabic*)]
			\item Show that ``being equivalent'' is an equivalence relation on the set of metrics on $M$ (reflexive, symmetric, transitive).
			\item Show that if $x_n\to x_0$ with respect to $d$, then $x_n\to x_0$ with respect to $\rho$.
		\end{enumerate}
	\end{exercise}
	
	\subsection{Continuity}
	
	Let $(X,d)$ and $(Y,\rho)$ be two metric spaces and let $f:X\to Y$ be a function.
	
	\begin{definition}{Continuity at a Point}
		For $x_0\in X$, we say that $f$ is \textbf{continuous at $x_0$} if
		\[
		\forall \varepsilon>0 \ \ \exists \delta_\varepsilon>0 \ \text{ s.t. } \ \forall x\in X \quad d(x,x_0)<\delta_\varepsilon \implies \rho\bigl(f(x),f(x_0)\bigr)<\varepsilon .
		\]
	\end{definition}
	
	\begin{definition}{Continuous Function}
		$f$ is called \textbf{continuous} if it is continuous at each $x_0\in X$, that is,
		\[
		\forall x_0\in X \ \ \forall \varepsilon>0 \ \ \exists \delta(\varepsilon,x_0)>0 \ \text{ s.t. } \ \forall x\in X \quad d(x,x_0)<\delta \implies \rho\bigl(f(x),f(x_0)\bigr)<\varepsilon .
		\]
	\end{definition}
	
	\begin{definition}{Uniform Continuity}
		$f$ is called \textbf{uniformly continuous} on $X$ if
		\[
		\forall \varepsilon>0 \ \ \exists \delta_\varepsilon>0 \ \text{ s.t. } \ \forall x',x''\in X \quad d(x',x'')<\delta_\varepsilon \implies \rho\bigl(f(x'),f(x'')\bigr)<\varepsilon .
		\]
	\end{definition}
	
	\begin{remark}
		In uniform continuity the number $\delta_\varepsilon$ depends only on $\varepsilon$, whereas in ordinary continuity it may also depend on the point $x_0$.
	\end{remark}
	
	\begin{definition}{Lipschitz Condition}
		A function $f:X\to Y$ is called \textbf{Lipschitz} if
		\[
		\exists L\in\R^+ \ \text{ s.t. } \ \rho\bigl(f(x'),f(x'')\bigr)\le L\, d(x',x'') \qquad \forall x',x''\in X .
		\]
		We write $f\in \mathrm{Lip}(X,Y)$.
	\end{definition}
	
	% Muhammed Alaa El-Din: I made the one-line proof explicit (delta = epsilon / L).
	\begin{remark}
		Every $f\in \mathrm{Lip}(X,Y)$ must be uniformly continuous: given $\varepsilon>0$, take $\delta_\varepsilon = \varepsilon/L$; then $d(x',x'')<\delta_\varepsilon$ implies $\rho\bigl(f(x'),f(x'')\bigr)\le L\,d(x',x'')<\varepsilon$.
	\end{remark}
	
	% Muhammed Alaa El-Din: I made the wording precise: f(x) = ax + b is affine (linear only when b = 0), and the admissible constants are exactly L >= |a| with L > 0 (L = |a| + 1 is just one choice, needed when a = 0).
	\begin{example}
		Any affine function $f:\R\to\R$, $f(x)=ax+b$, is Lipschitz with respect to the usual metric. Indeed,
		\[
		|f(x')-f(x'')| = |a(x'-x'')| = |a|\,|x'-x''| ,
		\]
		so any $L\ge |a|$ with $L>0$ works; for instance $L=|a|+1$ (the ``$+1$'' only matters when $a = 0$, since $L$ must be positive).
	\end{example}
	
	% Muhammed Alaa El-Din: I made the hypothesis "on an interval" explicit: the Mean Value Theorem needs the segment between x' and x'' to lie in the domain. On a non-interval domain the claim fails (example added).
	\begin{remark}
		If a function is differentiable \emph{on an interval} with bounded derivative, $|f'(x)|\le L$, then it is Lipschitz with constant $L$: by the Mean Value Theorem, $|f(x')-f(x'')| = |f'(c)|\,|x'-x''|\le L\,|x'-x''|$. The interval hypothesis matters: on $D = \left]0,1\right[\cup\left]1,2\right[$, the function equal to $0$ on $\left]0,1\right[$ and to $1$ on $\left]1,2\right[$ has $f' = 0$ but is not Lipschitz (take $x'' < 1 < x'$ arbitrarily close to $1$).
	\end{remark}
	
	\begin{exercise}
		Show that the function $f:\R\to\R$, $f(x)=x^2$, is \textbf{not} uniformly continuous.
	\end{exercise}
	
	  % Muhammed Alaa El-Din: I made this a "Solution" (it solves the exercise above) instead of an unlabeled proof, and made the final inequality match the negation (>= epsilon_0).
	  \begin{proof}[Solution]
    Recall the definition of uniform continuity for a function $f: X \to Y$:
    \[
    \forall \varepsilon > 0, \;\exists \delta > 0 \;\text{such that}\; \forall x', x'' \in X, \quad d(x', x'') < \delta \implies \rho(f(x'), f(x'')) < \varepsilon.
    \]
    To show that $f(x) = x^2$ is not uniformly continuous, we must prove the negation of this statement:
    \[
    \exists \varepsilon_0 > 0 \;\text{such that}\; \forall \delta > 0, \;\exists x', x'' \in \mathbb{R} \;\text{such that}\; |x' - x''| < \delta \;\land\; |f(x') - f(x'')| \ge \varepsilon_0.
    \]
    
    \noindent Choose $\varepsilon_0 = 1$. For any given $\delta > 0$, select:
    \[
    x' = \frac{1}{\delta}, \qquad x'' = \frac{1}{\delta} + \frac{\delta}{2}.
    \]
    
    \noindent First, we check the distance between $x'$ and $x''$:
    \[
    d(x', x'') = |x' - x''| = \left| \frac{1}{\delta} - \left(\frac{1}{\delta} + \frac{\delta}{2}\right) \right| = \frac{\delta}{2} < \delta.
    \]
    
    \noindent Next, we evaluate the distance between their images under $f(x) = x^2$:
    \begin{align*}
    \rho(f(x'), f(x'')) &= |f(x') - f(x'')| = |(x')^2 - (x'')^2| \\
    &= |x' - x''| \cdot |x' + x''| \\
    &= \left| \frac{1}{\delta} - \left(\frac{1}{\delta} + \frac{\delta}{2}\right) \right| \cdot \left| \frac{1}{\delta} + \left(\frac{1}{\delta} + \frac{\delta}{2}\right) \right| \\
    &= \frac{\delta}{2} \cdot \left| \frac{2}{\delta} + \frac{\delta}{2} \right| \\
    &= 1 + \frac{\delta^2}{4} \ge 1 = \varepsilon_0.
    \end{align*}
    
    \noindent Since we found $\varepsilon_0 = 1$ such that for every $\delta > 0$ there exist $x', x'' \in \mathbb{R}$ with $|x' - x''| < \delta$ and $|f(x') - f(x'')| \ge \varepsilon_0$, the function $f(x) = x^2$ is not uniformly continuous on $\mathbb{R}$.
    \end{proof}

	\subsection{Continuity in Terms of Open Balls}
	
	\begin{remark}
		Continuity at a point $x_0$ can be written in terms of open balls. The condition
		\[
		\forall \varepsilon>0 \ \ \exists \delta>0 \ \text{ s.t. } \ \forall x\in X \quad d(x,x_0)<\delta \implies \rho\bigl(f(x),f(x_0)\bigr)<\varepsilon
		\]
		is equivalent to
		\[
		\forall \varepsilon>0 \ \ \exists \delta>0 \ \text{ s.t. } \ \forall x\in X \quad x\in B_d(x_0;\delta) \implies x\in f^{-1}\bigl[B_\rho\bigl(f(x_0);\varepsilon\bigr)\bigr],
		\]
		that is,
		\[
		\forall \varepsilon>0 \ \ \exists \delta>0 \ \text{ s.t. } \ B_d(x_0;\delta)\subseteq f^{-1}\bigl[B_\rho\bigl(f(x_0);\varepsilon\bigr)\bigr].
		\]
	\end{remark}
	
	\subsection{Global Continuity}
	
	\begin{theorem}
		\label{thm:globalcont}
		A function $f:(X,d)\to(Y,\rho)$ is continuous if and only if the inverse image $f^{-1}[G]$ of every open set $G\subseteq Y$ is open in $X$.
	\end{theorem}
	
	% Muhammed Alaa El-Din: I made a proof for this theorem; it was stated without proof, yet it is used in Lecture 8 to prove that continuous images of compact sets are compact.
	\begin{proof}
		\textbf{($\Rightarrow$)} Let $G\subseteq Y$ be open and let $x_0\in f^{-1}[G]$. Then $f(x_0)\in G$, so there is $\varepsilon>0$ with $B_\rho(f(x_0);\varepsilon)\subseteq G$. By continuity at $x_0$ (in the form of the previous Remark) there is $\delta>0$ with
		\[
		B_d(x_0;\delta)\subseteq f^{-1}\bigl[B_\rho(f(x_0);\varepsilon)\bigr]\subseteq f^{-1}[G].
		\]
		So every point of $f^{-1}[G]$ is an interior point, i.e.\ $f^{-1}[G]$ is open.
		
		\medskip
		\noindent\textbf{($\Leftarrow$)} Let $x_0\in X$ and $\varepsilon>0$. The ball $B_\rho(f(x_0);\varepsilon)$ is open (Theorem~\ref{thm:openballopenset}), so by hypothesis $U = f^{-1}\bigl[B_\rho(f(x_0);\varepsilon)\bigr]$ is open in $X$. Since $x_0\in U$, there is $\delta>0$ with $B_d(x_0;\delta)\subseteq U$, which is continuity at $x_0$ by the previous Remark.
	\end{proof}

	\subsection{Sequential Continuity}
	
	\begin{theorem}[Sequential Continuity]
		\label{thm:seqcont}
		Let $f:(X,d)\to(Y,\rho)$ be a function and $x_0\in X$. Then
		\[
		f \text{ is continuous at } x_0 \iff \text{for any sequence } \{x_n\}_{n=1}^{\infty}\subseteq X \text{ with } x_n\to x_0, \ \ \lim_{n\to\infty} f(x_n)=f(x_0).
		\]
	\end{theorem}
	
	\begin{proof}
		\textbf{($\Rightarrow$)} If $f$ is continuous at $x_0$, then
		\[
		\forall \varepsilon>0 \ \ \exists \delta_\varepsilon>0 \ \text{ s.t. } \ \forall x\in X \quad d(x,x_0)<\delta_\varepsilon \implies \rho\bigl(f(x),f(x_0)\bigr)<\varepsilon . \tag{1}
		\]
		Since $x_n\to x_0$, for this $\delta_\varepsilon>0$ there is $N\in\Z^+$ such that
		\[
		\forall n\in\Z^+ \quad n\ge N \implies d(x_0,x_n)<\delta_\varepsilon . \tag{2}
		\]
		From (1) and (2) we have
		\[
		\forall \varepsilon>0 \ \ \exists N\in\Z^+ \ \text{ s.t. } \ \forall n\ge N \quad \rho\bigl(f(x_0),f(x_n)\bigr)<\varepsilon ,
		\]
		which means $\lim_{n\to\infty} f(x_n)=f(x_0)$.
		
		\medskip
		\noindent\textbf{($\Leftarrow$)} We prove the contrapositive. Suppose that $f$ is not continuous at $x_0$. Then
		\[
		\exists \varepsilon>0 \ \ \forall \delta>0 \ \ \exists x_\delta\in X \ \text{ s.t. } \ d(x_\delta,x_0)<\delta \ \text{ and } \ \rho\bigl(f(x_0),f(x_\delta)\bigr)\ge\varepsilon .
		\]
		Let $\delta_1=1,\ \delta_2=\tfrac12,\ \delta_3=\tfrac13$, and so on. Then there exist $x_1,x_2,x_3,\dots\in X$ such that
		\[
		d(x_n,x_0)<\tfrac1n \quad\text{and}\quad \rho\bigl(f(x_n),f(x_0)\bigr)\ge\varepsilon \qquad \forall n\in\Z^+ .
		\]
		Thus $x_n\to x_0$ but $f(x_n)\not\to f(x_0)$.
	\end{proof}
	
	\begin{example}
		Let
		\[
		f(x)=\begin{cases} \sin\!\left(\dfrac1x\right) & x\neq 0 \\[2mm] k & x=0 . \end{cases}
		\]
		Regardless of the value of $k$, this function is not continuous at $0$. Consider
		\[
		x_n=\frac{1}{2n\pi+\pi/2}\to 0, \qquad y_n=\frac{1}{n\pi}\to 0 .
		\]
		Then $f(x_n)=\sin\!\bigl(2n\pi+\tfrac{\pi}{2}\bigr)=1$ and $f(y_n)=\sin(n\pi)=0$ for all $n$. If $f$ were continuous at $0$, Theorem~\ref{thm:seqcont} would give $f(x_n)\to f(0)=k$ and $f(y_n)\to k$, hence $k=1$ and $k=0$, a contradiction.
	\end{example}
	
	\subsection{Uniform Continuity and Cauchy Sequences}
	
	\begin{theorem}
		Uniformly continuous functions preserve Cauchy sequences. That is, if $f:(X,d)\to(Y,\rho)$ is uniformly continuous and $\{x_n\}_{n=1}^{\infty}$ is Cauchy in $X$, then $\{f(x_n)\}_{n=1}^{\infty}$ is Cauchy in $Y$.
	\end{theorem}
	
	\begin{proof}
		Since $f$ is uniformly continuous,
		\[
		\forall \varepsilon>0 \ \ \exists \delta_\varepsilon>0 \ \text{ s.t. } \ \forall x',x''\in X \quad d(x',x'')<\delta_\varepsilon \implies \rho\bigl(f(x'),f(x'')\bigr)<\varepsilon . \tag{1}
		\]
		Since $\{x_n\}\subseteq X$ is Cauchy, for the number $\delta_\varepsilon>0$ there is $N\in\Z^+$ such that
		\[
		\forall m,n\ge N \quad d(x_n,x_m)<\delta_\varepsilon . \tag{2}
		\]
		From (1) and (2),
		\[
		\forall \varepsilon>0 \ \ \exists N\in\Z^+ \ \text{ s.t. } \ \forall m,n\ge N \quad \rho\bigl(f(x_n),f(x_m)\bigr)<\varepsilon ,
		\]
		so $\{f(x_n)\}_{n=1}^{\infty}$ is Cauchy in $Y$.
	\end{proof}
	
	% Muhammed Alaa El-Din: I made the cited facts explicit (completeness of R and Theorem "Sequential Continuity"), and added that mere continuity does NOT preserve Cauchy sequences in general, so that the role of completeness is clear.
	\begin{caution}

The function $f(x)=x^2$ is not uniformly continuous on $\R$, but it preserves Cauchy sequences: a Cauchy sequence in $\R$ converges (completeness of $\R$), and by Theorem~\ref{thm:seqcont} a continuous function maps a convergent sequence to a convergent sequence, which is Cauchy.

		\noindent On a non-complete space, a merely continuous function need not preserve Cauchy sequences: $f(x) = \tfrac1x$ on $\left]0,1\right[$ is continuous, $x_n = \tfrac1{n+1}$ is Cauchy, but $f(x_n) = n+1$ is not.
	\end{caution}
	
	\subsection{Compactness}
	
	\begin{definition}{Open Cover}
		A collection $\{G_i : i\in I\}$ of open sets is called an \textbf{open cover} for the set $K$ if
		\[
		K\subseteq \bigcup_{i\in I} G_i .
		\]
		(More generally, $\{G_i:i\in I\}$ is a \textbf{cover} for $A$ if $A\subseteq\bigcup_{i\in I}G_i$.)
	\end{definition}
	
	% Muhammed Alaa El-Din: I made the set in this definition explicit (the original switched from K to A without saying which set is covered).
	\begin{definition}{Subcover}
		Let $\{G_i:i\in I\}$ be a cover for a set $A$. By a \textbf{subcover} of $\{G_i:i\in I\}$ we mean a collection $\{G_j:j\in J\}$ where $J\subseteq I$ which also covers $A$, that is,
		\[
		A\subseteq\bigcup_{j\in J} G_j .
		\]
	\end{definition}
	
	\begin{definition}{Compact Set}
		A subset $K$ of a metric space $(X,d)$ is called \textbf{compact} if every open cover of $K$ has a finite subcover. That is, whenever
		\[
		K\subseteq\bigcup_{i\in I} G_i \quad (G_i \text{ open}),
		\]
		there exist $i_1,\dots,i_n\in I$ such that
		\[
		K\subseteq G_{i_1}\cup G_{i_2}\cup\cdots\cup G_{i_n} .
		\]
	\end{definition}
	
	\begin{theorem}[Compact sets are bounded]
		Every compact set is bounded.
	\end{theorem}
	
	\begin{proof}
		Let $K$ be compact. For every $x\in K$ we have $x\in B(x;1)$, so
		\[
		K\subseteq\bigcup_{x\in K}B(x;1)
		\]
		is an open cover of $K$ (open balls are open sets). By compactness there exist $x_1,x_2,\dots,x_m\in K$ such that
		\[
		K\subseteq\bigcup_{i=1}^{m}B(x_i;1) .
		\]
		A finite union of bounded sets is bounded (Theorem~\ref{thm:unionbounded}), hence $K$ is bounded.
	\end{proof}
	
	\begin{theorem}[Compact sets are closed]
		Every compact set is closed.
	\end{theorem}
	
	\begin{proof}
		Let $K$ be a compact set and let $y\in K^c$. For every $x\in K$ we have $x\neq y$, so we can find $\varepsilon_x>0$ (for instance $\varepsilon_x=\tfrac12 d(x,y)$) such that
		\[
		B(x;\varepsilon_x)\cap B(y;\varepsilon_x)=\emptyset .
		\]
		Now $K\subseteq\bigcup_{x\in K}B(x;\varepsilon_x)$ is an open cover of $K$. Since $K$ is compact, there are $x_1,\dots,x_n\in K$ with
		\[
		K\subseteq\bigcup_{i=1}^{n}B(x_i;\varepsilon_{x_i}) .
		\]
		Take $\varepsilon=\min\{\varepsilon_{x_1},\dots,\varepsilon_{x_n}\}>0$. Then $B(y;\varepsilon)\subseteq B(y;\varepsilon_{x_i})$ is disjoint from $B(x_i;\varepsilon_{x_i})$ for every $i$, so
		\[
		B(y;\varepsilon)\cap K=\emptyset \implies B(y;\varepsilon)\subseteq K^c .
		\]
		Thus every point of $K^c$ is an interior point, so $K^c$ is open and $K$ is closed.
	\end{proof}
	
	\begin{theorem}
		A closed subset of a compact set is compact.
	\end{theorem}
	
	\begin{proof}
		Let $K$ be compact and let $F\subseteq K$ be closed. Let $\{G_i:i\in I\}$ be an open cover for $F$. Add $F^c$ (which is open) to this collection; the new collection is an open cover for $K$. Hence it has a finite subcover for $K$. If this finite subcover contains $F^c$, remove it. What remains is a finite subcollection of $\{G_i\}$ which still covers $F$ (since $F^c\cap F=\emptyset$). So $F$ is compact.
	\end{proof}
	
	\begin{theorem}[Closed intervals are compact]
		\label{thm:closedintervalcompact}
		Any closed interval $I=[a,b]$ is compact in $\R$.
	\end{theorem}
	
	% Muhammed Alaa El-Din: I made the last step of this proof complete. The original ended with "the same argument shows b \in S"; but beta_0 = b only says sup S = b, not that b itself belongs to S. I wrote out the argument. I also removed the unused remark about [a, beta[ \subseteq G_{i_0} and spelled out why [beta, beta_0 + eta/2] is covered by G_j.
	\begin{proof}
		Let $\{G_i:i\in I_0\}$ be an open cover for $[a,b]$. Define
		\[
		S=\bigl\{\beta\in[a,b] : [a,\beta] \text{ is covered by finitely many of the } G_i\bigr\}, \qquad \beta_0=\sup S .
		\]
		The set $S$ is non-empty ($a\in S$, since $a$ lies in some $G_{i_0}$, which alone covers $[a,a]$) and bounded above by $b$, so $\beta_0$ exists and $\beta_0\in[a,b]$.
		
		\smallskip
		\noindent \emph{Key step.} The point $\beta_0$ lies in some $G_j$, which is open, so $]\beta_0-\eta,\beta_0+\eta[\,\subseteq G_j$ for some $\eta>0$. By the definition of supremum there is $\beta\in S$ with $\beta_0-\eta<\beta\le\beta_0$. Then $[a,\beta]$ is covered by finitely many $G_i$'s, and
		\[
		[\beta,\ \beta_0+\eta/2] \subseteq\ ]\beta_0-\eta,\beta_0+\eta[\ \subseteq G_j ,
		\]
		so adding $G_j$ gives a finite subcover of $[a,\beta_0+\eta/2]\cap[a,b]$. Hence $\min\{\beta_0+\eta/2,\,b\}\in S$.
		
		\smallskip
		\noindent \emph{$\beta_0 = b$.} If $\beta_0<b$, then $\min\{\beta_0+\eta/2,\,b\}>\beta_0$ belongs to $S$, contradicting $\beta_0=\sup S$.
		
		\smallskip
		\noindent \emph{$b\in S$.} Since $\beta_0 = b$, the key step gives $\min\{b+\eta/2,\,b\} = b\in S$. So $[a,b]$ is covered by finitely many $G_i$, i.e.\ the cover has a finite subcover.
	\end{proof}
	
	\begin{theorem}
		A bounded closed subset of $\R$ is compact.
	\end{theorem}
	
	\begin{proof}
		If $K\subseteq\R$ is bounded, then $K\subseteq[a,b]$ for some $a<b$. Since $K$ is a closed subset of the compact set $[a,b]$ (Theorem~\ref{thm:closedintervalcompact}), it is compact.
	\end{proof}
	
	\begin{takeaway}
		In $\R$: \ $K$ is compact $\iff$ $K$ is closed and bounded.
	\end{takeaway}

	\subsection{Compact Subsets of $\R$}
	
	\begin{recall}[Supremum]
		$\beta=\sup K$ if and only if
		\begin{enumerate}[label=(\roman*)]
			\item $x\le\beta$ for all $x\in K$, and
			\item $\forall\varepsilon>0 \ \ \exists x_\beta\in K$ such that $\beta-\varepsilon<x_\beta\le\beta$.
		\end{enumerate}
	\end{recall}
	
	\begin{theorem}
		\label{thm:maxmin}
		A non-empty compact subset of $\R$ has a maximum (and a minimum).
	\end{theorem}
	
	% Muhammed Alaa El-Din: I made the link between the hypothesis (compact) and the proof (closed and bounded) explicit, and cited the completeness axiom for the existence of sup and inf.
	\begin{proof}
		Let $K\neq\emptyset$ be compact. By Lecture~7, $K$ is bounded and closed. By the completeness axiom of $\R$, $\alpha=\inf K$ and $\beta=\sup K$ exist. If $\beta\in K$ we are done. Otherwise, by property (ii) of the supremum, for every $\varepsilon>0$ there is $x_\beta\in K$ with $\beta-\varepsilon<x_\beta<\beta$, so $x_\beta\neq\beta$ and $\beta\in K'$ (accumulation point). Since $K$ is closed,
		\[
		\beta\in K'\subseteq K ,
		\]
		which contradicts $\beta\notin K$. Hence $\beta\in K$, that is, $\beta=\max K$. The same argument gives $\alpha=\min K$.
	\end{proof}
	
	\subsection{Continuous Images of Compact Sets}
	
	\begin{theorem}
		\label{thm:imagecompact}
		The direct image of a compact set by a continuous function is compact.
	\end{theorem}
	
	\begin{proof}
		Let $f:(X,d)\to(Y,\rho)$ be continuous and let $K\subseteq X$ be compact. We prove that $f[K]$ is compact in $Y$.
		
		Let $\{G_i:i\in I\}$ be an open cover for $f[K]$, i.e.
		\[
		f[K]\subseteq\bigcup_{i\in I}G_i .
		\]
		Then
		\[
		K\subseteq f^{-1}\Bigl[\bigcup_{i\in I}G_i\Bigr]=\bigcup_{i\in I}f^{-1}[G_i] .
		\]
		Each $f^{-1}[G_i]$ is open in $X$ since $f$ is continuous (Theorem~\ref{thm:globalcont}), so $\{f^{-1}[G_i]:i\in I\}$ is an open cover for $K$. By compactness of $K$, there exist $i_1,\dots,i_n\in I$ such that
		\[
		K\subseteq\bigcup_{k=1}^{n}f^{-1}[G_{i_k}] .
		\]
		% Muhammed Alaa El-Din: I made the last step justified, using f[f^{-1}[G]] \subseteq G.
		Then, since $f\bigl[f^{-1}[G]\bigr]\subseteq G$ for every $G\subseteq Y$,
		\[
		f[K]\subseteq\bigcup_{k=1}^{n}f\bigl[f^{-1}[G_{i_k}]\bigr]\subseteq\bigcup_{k=1}^{n}G_{i_k} .
		\]
		Hence $f[K]$ is compact.
	\end{proof}
	
	\begin{corollary}[Extreme Value Theorem]
		Let $(M,d)$ be a metric space and let $\emptyset\neq K\subseteq M$ be compact. If $f:M\to\R$ is continuous, then $f$ has a maximum (and a minimum) on $K$. That is, there exist $x_1,x_2\in K$ such that
		\[
		f(x_1)\le f(x)\le f(x_2) \qquad \forall x\in K .
		\]
	\end{corollary}
	
	\begin{proof}
		By Theorem~\ref{thm:imagecompact}, $f[K]$ is a non-empty compact subset of $\R$, so by Theorem~\ref{thm:maxmin} it has a minimum $f(x_1)$ and a maximum $f(x_2)$ for some $x_1,x_2\in K$.
	\end{proof}

\subsection{Finite Intersection Property}

% Muhammed Alaa El-Din: I made the wording of this definition non-redundant (it said "the intersection ... has a non-empty intersection").
\begin{definition}{Finite Intersection Property}
	A collection of sets is said to have the \textbf{finite intersection property} if every finite subcollection has a non-empty intersection.
\end{definition}

% Muhammed Alaa El-Din: I made the theorem state the standard (general) characterization for arbitrary families of closed sets and gave it a proof; the original stated only the nested-sequence version, with no proof. Its "if" direction for sequences is true in metric spaces but much harder (it is equivalent to sequential compactness), so I kept the nested version as a corollary in the direction that follows directly, and flagged the converse.
\begin{theorem}
	\label{thm:fip}
	A metric space $M$ is compact if and only if every collection of closed subsets of $M$ with the finite intersection property has a non-empty intersection.
\end{theorem}

\begin{proof}
	For a collection $\{F_i\}_{i\in I}$ of closed sets put $G_i = F_i^c$, which are open. By De Morgan,
	\[
	\bigcap_{i\in I} F_i = \emptyset \iff \bigcup_{i\in I} G_i = M,
	\qquad
	\bigcap_{k=1}^{n} F_{i_k} = \emptyset \iff \bigcup_{k=1}^{n} G_{i_k} = M .
	\]
	
	\noindent ($\Rightarrow$) Let $M$ be compact and let $\{F_i\}$ be closed sets with the finite intersection property. If $\bigcap_i F_i = \emptyset$, then $\{G_i\}$ is an open cover of $M$, so it has a finite subcover $G_{i_1},\dots,G_{i_n}$, i.e.\ $F_{i_1}\cap\cdots\cap F_{i_n}=\emptyset$, contradicting the finite intersection property.
	
	\medskip
	\noindent ($\Leftarrow$) Let $\{G_i\}$ be an open cover of $M$ and suppose it has no finite subcover. Then for every finite choice $i_1,\dots,i_n$ we have $\bigcup_k G_{i_k}\neq M$, i.e.\ $\bigcap_k F_{i_k}\neq\emptyset$. So the closed sets $F_i = G_i^c$ have the finite intersection property, and by hypothesis $\bigcap_i F_i\neq\emptyset$, i.e.\ $\bigcup_i G_i\neq M$, contradicting that $\{G_i\}$ covers $M$.
\end{proof}

\begin{corollary}[Nested Closed Sets]
	Let $M$ be compact and let
	\[
	A_1 \supseteq A_2 \supseteq A_3 \supseteq \cdots
	\]
	be a nested sequence of non-empty closed sets. Then
	\[
	\bigcap_{n=1}^{\infty} A_n \neq \emptyset .
	\]
\end{corollary}

\begin{proof}
	For a nested sequence the finite intersection property is just the statement that each $A_n$ is non-empty:
	\[
	\bigcap_{k=1}^{n} A_k = A_n \neq \emptyset \qquad \forall n .
	\]
	Apply Theorem~\ref{thm:fip}. Passing to complements (De Morgan), the conclusion reads
	\[
	\Bigl(\bigcap_{n=1}^{\infty} A_n\Bigr)^{c} = \bigcup_{n=1}^{\infty} A_n^{c} \neq M . \qedhere
	\]
\end{proof}

\begin{remark}
	The converse also holds in metric spaces: if every nested sequence of non-empty closed sets in $M$ has a non-empty intersection, then $M$ is compact. Its proof is harder (it goes through sequential compactness) and is omitted here.
\end{remark}

\subsection{Dense Sets}

\begin{definition}{Dense Set}
	A set $D$ of a metric space $M$ is called \textbf{dense} if $\cl{D} = M$.
\end{definition}

\noindent Equivalently (by Theorem~\ref{thm:closure}), $D$ is dense if every element $x\in M$ can be approximated by a sequence $\{x_n\}_{n=1}^{\infty}\subseteq D$, that is,
\[
\forall x\in M \quad \exists \{x_n\}_{n=1}^{\infty}\subseteq D \ \text{ s.t. } \ x_n \to x .
\]

% Muhammed Alaa El-Din: I made the approximation work for every real x (the original only treated x in [0,1], which does not prove density in all of R), renamed the decimal digits to d_k (the letter x_n was also used for sequences), and added the key inequality as a hint for the assignment.
\begin{example}[Rational numbers]
	$\Q$ is dense in $\R$ (with the usual metric).
	
	\smallskip
	\noindent\textbf{Decimal approximations.} For $x\in[0,1[$ write its decimal expansion
	\[
	x = 0.d_1d_2d_3\cdots d_n\cdots = \sum_{k=1}^{\infty}\frac{d_k}{10^k}.
	\]
	For an arbitrary $x\in\R$ take
	\[
	y_n = \frac{\lfloor 10^n x\rfloor}{10^n}.
	\]
	Then $y_n\in\Q$ for all $n\in\Z^+$ (for $x\in[0,1[$, $\lfloor 10x\rfloor = d_1$, $\lfloor 100x\rfloor = 10d_1+d_2$, and so on, so $y_n = 0.d_1d_2\cdots d_n$ is the truncated expansion), and
	\[
	\lim_{n\to\infty} y_n = x \qquad \textit{(Assignment. Hint: } 0\le x-y_n<10^{-n}.)
	\]
\end{example}

\end{remark}

\subsection{Bernstein Polynomials}

\noindent The map
\[
[0,1]\to[a,b], \qquad x\mapsto (b-a)x+a
\]
allows us to work on $[0,1]$ without loss of generality.

\begin{definition}{Bernstein Polynomial}
	Given any continuous function $f$ on $[0,1]$, define
	\[
	B_n(f;x) = \sum_{k=0}^{n}\binom{n}{k} f\!\left(\frac{k}{n}\right) x^{k}(1-x)^{n-k}.
	\]
\end{definition}

% Muhammed Alaa El-Din: I made the notation consistent (the norm ||.||_\infty was never defined; it is d_\infty), named the theorem (it proves the Weierstrass Approximation Theorem), and added the transfer from [0,1] to [a,b], which the "In particular" needs.
\begin{theorem}[Bernstein; Weierstrass Approximation Theorem]
	For every continuous $f$ on $[0,1]$,
	\[
	\forall \varepsilon>0 \ \ \exists N_\varepsilon \ \text{ s.t. } \ \forall n\ge N_\varepsilon \quad d_\infty\bigl(f, B_n(f)\bigr) < \varepsilon .
	\]
	In particular, the polynomials are dense in $\bigl(C[a,b],d_\infty\bigr)$.
\end{theorem}

\textit{Proof: to be done as an assignment.}

\noindent \textit{Transfer to $[a,b]$.} Given $g\in C[a,b]$ and $\varepsilon>0$, the function $f(x)=g\bigl((b-a)x+a\bigr)$ is continuous on $[0,1]$, so some polynomial $p$ satisfies $d_\infty(f,p)<\varepsilon$ on $[0,1]$. Then $q(t)=p\bigl(\tfrac{t-a}{b-a}\bigr)$ is a polynomial and $d_\infty(g,q)<\varepsilon$ on $[a,b]$, since $t\mapsto\tfrac{t-a}{b-a}$ maps $[a,b]$ onto $[0,1]$.

\subsection{Total Boundedness}

\begin{definition}{Totally Bounded}
	Let $(M,d)$ be a metric space. A subset $A$ of $M$ is called \textbf{totally bounded} if for each $\varepsilon>0$ there is a finite set $\{x_1,\dots,x_n\}\subseteq M$ such that
	\[
	A\subseteq\bigcup_{k=1}^{n}B(x_k;\varepsilon).
	\]
	In this case, the set $\{x_1,\dots,x_n\}$ is called an \textbf{$\varepsilon$-net} for $A$.
\end{definition}

% Muhammed Alaa El-Din: I made the claims in this remark justified: short proofs of "compact => totally bounded" and "totally bounded => bounded", and counterexamples for both converses (they were asserted without examples).
\begin{remark}
	\begin{itemize}
		\item \textbf{Compact $\Rightarrow$ totally bounded.} If $A$ is compact and $\varepsilon>0$, the open cover $\{B(x;\varepsilon):x\in A\}$ of $A$ has a finite subcover, which is an $\varepsilon$-net. The converse is not true: $\oi{0}{1}$ is totally bounded in $\R$ but not compact (it is not closed).
		\item \textbf{Totally bounded $\Rightarrow$ bounded.} Take $\varepsilon=1$: $A$ lies in a finite union of balls, which is bounded (Theorem~\ref{thm:unionbounded}). The converse is not true: an infinite set with the discrete metric is bounded (it lies in any ball of radius $2$), but no finite family of balls of radius $\tfrac12$ (each a single point) covers it.
	\end{itemize}
\end{remark}

\subsection{Completion of a Metric Space}

\begin{definition}{Isometry}
	By an \textbf{isometry} between two metric spaces $(X,d)$ and $(Y,\rho)$ we mean a function $\phi:X\to Y$ that preserves the distance, that is,
	\[
	\rho\bigl(\phi(x),\phi(x')\bigr) = d(x,x') \qquad \forall x,x'\in X .
	\]
	(An isometry is automatically injective: $\phi(x)=\phi(x')$ gives $d(x,x')=0$.)
\end{definition}

\begin{definition}{Cauchy Completion}
	By a \textbf{Cauchy completion} of a metric space $(X,d)$ we mean a complete metric space $(X^*,d^*)$ and an isometry
	\[
	\phi:X\to X^*
	\]
	such that $\phi(X)$ is dense in $X^*$.
\end{definition}

\subsection{The Cauchy Completion}

Let $(X,d)$ be a metric space and let $C(X)$ be the set of all Cauchy sequences in $X$.

\begin{definition}{Equivalent Cauchy Sequences}
	Let $\{x_n\}$ and $\{x_n'\}$ be two Cauchy sequences in $X$. We say that $\{x_n\}$ and $\{x_n'\}$ are \textbf{equivalent} if
	\[
	\lim_{n\to\infty} d(x_n,x_n') = 0 .
	\]
\end{definition}

% Muhammed Alaa El-Din: I made the set correct: the relation is defined on C(X), the set of Cauchy sequences; X^* is the set of its equivalence classes and is only defined afterwards.
\begin{exercise}[Assignment]
	Show that this is an equivalence relation on $C(X)$ (and that the limit above exists).
\end{exercise}

\noindent Let $X^*$ be the set of all equivalence classes of Cauchy sequences in $X$:
\[
X^* = \Bigl\{\,\bigl[\{x_n\}\bigr] : \{x_n\} \text{ is a Cauchy sequence in } X\,\Bigr\}.
\]
We define $d^*$ on $X^*$ by
\[
d^*\Bigl(\bigl[\{x_n\}\bigr],\bigl[\{y_n\}\bigr]\Bigr) = \lim_{n\to\infty} d(x_n,y_n).
\]

\begin{exercise}[Assignment]
	Prove that $d^*$ is a well-defined metric on $X^*$. That is:
	\begin{enumerate}[label=(\arabic*)]
		\item the limit exists;
		\item it does not depend on the choice of representatives of the classes (well-definedness);
		\item $d^*$ satisfies the three metric axioms.
	\end{enumerate}
\end{exercise}

\begin{takeaway}
	Next, we then prove that $(X^*,d^*)$ is complete, and that $X$ is isometric to a dense subset of $X^*$.
\end{takeaway}

\subsection{The Equivalence Relation on Cauchy Sequences}

Let $(X,d)$ be a metric space and let $C(X)$ be the set of all Cauchy sequences in $X$. Define a relation $\sim$ on $C(X)$ by
\[
\{x_n\}\sim\{x_n'\} \iff \lim_{n\to\infty} d(x_n,x_n')=0 .
\]

\begin{proposition}
	The relation $\sim$ is an equivalence relation on $C(X)$.
\end{proposition}

\begin{proof}
	\textbf{(i) Reflexive.} $d(x_n,x_n)=0\to 0$, so $\{x_n\}\sim\{x_n\}$.
	
	\smallskip
	\noindent\textbf{(ii) Symmetric.} $d(x_n,x_n')=d(x_n',x_n)$, so if $d(x_n,x_n')\to0$ then $d(x_n',x_n)\to 0$.
	
	\smallskip
	\noindent\textbf{(iii) Transitive.} Suppose $\{x_n\} \sim\{x_n'\}$ and $\{x_n'\}\sim\{x_n''\}$. Then $d(x_n,x_n')\to0$ and $d(x_n',x_n'')\to0$. Since
	\[
	0\le d(x_n,x_n'')\le d(x_n,x_n')+d(x_n',x_n'') ,
	\]
	the right-hand side tends to $0$, so $d(x_n,x_n'')\to 0$. Hence $\{x_n\}\sim\{x_n''\}$.
\end{proof}

\subsection{The Metric Space $(X^{*},d^{*})$}

Let $X^{*}$ be the set of all equivalence classes under $\sim$:
\[
X^{*} = \Bigl\{\,\bigl[\{x_n\}\bigr] : \{x_n\} \text{ is a Cauchy sequence in } X\,\Bigr\}.
\]
We define $d^{*}$ on $X^{*}$ by
\[
d^{*}\Bigl(\bigl[\{x_n\}\bigr],\bigl[\{y_n\}\bigr]\Bigr) := \lim_{n\to\infty} d(x_n,y_n).
\]

\begin{proposition}
	The limit above exists, and it does not depend on the choice of the representatives of the classes.
\end{proposition}

\begin{proof}
	\textbf{Existence of the limit.} We show that $\{d(x_n,y_n)\}_{n}$ is a Cauchy sequence in $\R$. By the triangle inequality, for all $m,n$,
	\[
	d(x_n,y_n)\le d(x_n,x_m)+d(x_m,y_m)+d(y_m,y_n),
	\]
	and exchanging the roles of $(n,m)$,
	\[
	d(x_m,y_m)\le d(x_m,x_n)+d(x_n,y_n)+d(y_n,y_m).
	\]
	Hence
	\[
	\bigl|d(x_n,y_n)-d(x_m,y_m)\bigr|\le d(x_n,x_m)+d(y_n,y_m).
	\]
	Given $\varepsilon>0$, since $\{x_n\}$ and $\{y_n\}$ are Cauchy there is $N$ such that $d(x_n,x_m)<\varepsilon/2$ and $d(y_n,y_m)<\varepsilon/2$ for all $m,n\ge N$. So $|d(x_n,y_n)-d(x_m,y_m)|<\varepsilon$ for $m,n\ge N$, i.e.\ $\{d(x_n,y_n)\}$ is Cauchy in $\R$. Since $\R$ is complete, the limit exists.
	
	\medskip
	\noindent\textbf{Independence of the representatives.} Let $\{x_n\}\sim\{x_n'\}$ and $\{y_n\}\sim\{y_n'\}$. Look at these two inequalities:
	\begin{align*}
		d(x_n,y_n)&\le d(x_n,x_n')+d(x_n',y_n')+d(y_n',y_n),\\
		d(x_n',y_n')&\le d(x_n',x_n)+d(x_n,y_n)+d(y_n,y_n').
	\end{align*}
	Together they give
	\[
	\bigl|d(x_n,y_n)-d(x_n',y_n')\bigr|\le d(x_n,x_n')+d(y_n,y_n')\to 0.
	\]
	Therefore $\lim d(x_n,y_n)=\lim d(x_n',y_n')$, so the limit does not depend on the representatives.
\end{proof}

\noindent It remains to prove that $d^{*}$ satisfies the three conditions of a metric.

\begin{proposition}
	$d^{*}$ is a metric on $X^{*}$.
\end{proposition}

\begin{proof}
	\textbf{(i) Non-negativity and definiteness.} Non-negativity is obvious, being a limit of non-negative numbers. If
	\[
	d^{*}\Bigl(\bigl[\{x_n\}\bigr],\bigl[\{x_n'\}\bigr]\Bigr)=0,
	\]
	then $\lim d(x_n,x_n')=0$, that is, $\bigl[\{x_n\}\bigr]=\bigl[\{x_n'\}\bigr]$. Conversely, equal classes have distance $0$.
	
	\smallskip
	\noindent\textbf{(ii) Symmetry.} $d(x_n,y_n)=d(y_n,x_n)$ for every $n$.
	
	\smallskip
	\noindent\textbf{(iii) Triangle inequality.} See the exercise below.
\end{proof}

\subsection{Embedding $X$ into $X^{*}$}

For each $x\in X$ define
\[
\widehat{x}=\Bigl[\{x\}_{n=1}^{\infty}\Bigr]
\]
the class of the constant sequence $x,x,x,\dots$ (a constant sequence is clearly Cauchy). Define
\[
\phi:X \to X^{*}, \qquad \phi(x)=\widehat{x}.
\]

\begin{proposition}
	$\phi$ is an isometry.
\end{proposition}

\begin{proof}
	For $x,y\in X$,
	\[
	d^{*}(\widehat x,\widehat y)=\lim_{n\to\infty}d(x,y)=d(x,y). 
	\]
	Hence $\phi$ preserves distances.
\end{proof}

\subsection{$\phi(X)$ is Dense in $X^{*}$}

\begin{proposition}
	$\phi(X)$ is dense in $X^{*}$.
\end{proposition}

% Muhammed Alaa El-Din: I made the last step explicit: "every ball around x^* meets phi(X)" gives x^* in the closure via the closure--sequence theorem (take epsilon = 1/k).
\begin{proof}
	Let $x^{*}=\bigl[\{x_n\}\bigr]\in X^{*}$, where $\{x_n\}$ is Cauchy, and let $\varepsilon>0$. There is $N_\varepsilon\in\Z^+$ such that
	\[
	\forall\, m,n\ge N_\varepsilon \quad d(x_n,x_m)<\varepsilon/2 .
	\]
	In particular $d(x_m,x_{N_\varepsilon})<\varepsilon/2$ for all $m\ge N_\varepsilon$. Consider the constant sequence $x_{N_\varepsilon},x_{N_\varepsilon},\dots$, that is, the element $\widehat{x_{N_\varepsilon}}=\phi(x_{N_\varepsilon})\in\phi(X)$. Then
	\[
	d^{*}\bigl(x^{*},\widehat{x_{N_\varepsilon}}\bigr)=\lim_{m\to\infty}d(x_m,x_{N_\varepsilon})\le\frac{\varepsilon}{2}<\varepsilon .
	\]
	So for every element $x^{*}\in X^{*}$ and every $\varepsilon>0$ there is $\widehat x\in\phi(X)$ with $d^{*}(x^{*},\widehat x)<\varepsilon$. Taking $\varepsilon=\tfrac1k$ ($k=1,2,\dots$) gives a sequence in $\phi(X)$ converging to $x^{*}$, so $x^{*}\in\overline{\phi(X)}$ by Theorem~\ref{thm:closure}. Hence $\overline{\phi(X)}=X^{*}$.
\end{proof}

\subsection{$X^{*}$ is Complete}

It remains to prove that $(X^{*},d^{*})$ is complete.

\begin{proposition}
	Every Cauchy sequence in the dense subset $\phi(X)$ converges in $X^{*}$.
\end{proposition}

\begin{proof}
	Let $\{\widehat{z_k}\}_{k}$ be a Cauchy sequence in $\phi(X)$. Since $\phi$ is an isometry,
	\[
	d(z_n,z_m)=d^{*}(\widehat{z_n},\widehat{z_m})\qquad\forall\, m,n\in\Z^+ ,
	\]
	hence $\{z_1,z_2,z_3,\dots\}$ is a Cauchy sequence in $X$. For $\varepsilon>0$ there is $N_\varepsilon\in\Z^+$ such that
	\[
	d(z_k,z_n)<\frac{\varepsilon}{2}\qquad\text{for } k,n\ge N_\varepsilon .
	\]
	Let $z^{*}=\bigl[\{z_1,z_2,\dots\}\bigr]\in X^{*}$. For $k\ge N_\varepsilon$,
	\[
	d^{*}\bigl(\widehat{z_k},z^{*}\bigr)=\lim_{n\to\infty}d(z_k,z_n)\le\frac{\varepsilon}{2}<\varepsilon .
	\]
	Thus $\widehat{z_k}\to z^{*}$.
\end{proof}


\begin{theorem}
	$(X^{*},d^{*})$ is complete.
\end{theorem}

% Muhammed Alaa El-Din: I made the step "so {\hat z_k} is Cauchy" explicit with an epsilon/3 argument; the original only displayed the inequality.
\begin{proof}
	Let $\{w_k\}$ be any Cauchy sequence in $X^{*}$. Since $\phi(X)$ is dense, for each $k$ there is $z_k\in X$ with $d^{*}(w_k,\widehat{z_k})<\tfrac1k$. Then
	\[
	d^{*}(\widehat{z_k},\widehat{z_m})\le d^{*}(\widehat{z_k},w_k)+d^{*}(w_k,w_m)+d^{*}(w_m,\widehat{z_m})<\frac1k+d^{*}(w_k,w_m)+\frac1m .
	\]
	Given $\varepsilon>0$, choose $N$ with $\tfrac1N<\tfrac{\varepsilon}{3}$ and $d^{*}(w_k,w_m)<\tfrac{\varepsilon}{3}$ for all $k,m\ge N$; then $d^{*}(\widehat{z_k},\widehat{z_m})<\varepsilon$ for $k,m\ge N$. So $\{\widehat{z_k}\}$ is Cauchy in $\phi(X)$. By the previous proposition $\widehat{z_k}\to z^{*}$ for some $z^{*}\in X^{*}$. Finally
	\[
	d^{*}(w_k,z^{*})\le d^{*}(w_k,\widehat{z_k})+d^{*}(\widehat{z_k},z^{*})<\frac1k+d^{*}(\widehat{z_k},z^{*})\xrightarrow[k\to\infty]{}0 ,
	\]
	so $w_k\to z^{*}$. Hence every Cauchy sequence in $X^{*}$ converges.
\end{proof}

% Muhammed Alaa El-Din: I made a closing summary tying the results of this lecture back to the definition of a Cauchy completion in Lecture 10.
\begin{takeaway}
	Putting everything together: $(X^{*},d^{*})$ is a complete metric space, $\phi:X\to X^{*}$ is an isometry, and $\phi(X)$ is dense in $X^{*}$. Hence $(X^{*},d^{*})$ together with $\phi$ is a Cauchy completion of $(X,d)$. In particular, every metric space has a completion.
\end{takeaway}

\subsection{Pigeonhole Principle and Equipotency}

\begin{theorem}[Pigeonhole Principle]
	If $n>m$ ($n,m\in\Z^+$) and $n$ objects are put inside $m$ containers, then at least one container has more than one object.
\end{theorem}

\noindent In terms of sets: if $A$ and $B$ are finite sets with $|A|>|B|$, there is no injective function $A\to B$.

\begin{definition}{Equipotent Sets}
	We call two sets $A$ and $B$ \textbf{equipotent} (and write $A \approx  B$) if there is a bijection $f:A\to B$.
\end{definition}

\begin{exercise}
	Show that equipotency is an equivalence relation (reflexive, symmetric, transitive).
\end{exercise}


\subsection{Infinite Sets}

According to the pigeonhole principle, no finite set is equipotent to a proper subset of itself. (If $A$ has $n$ elements and $B\subset A$ has $m<n$ elements, an injection $A\to B$ would put $n$ objects in $m$ containers with no container used twice.) Thus we can define infinite sets as follows.

\begin{definition}{Infinite Set}
	A set $A$ is \textbf{infinite} if it is equipotent to one of its proper subsets.
\end{definition}

% Muhammed Alaa El-Din: I made a remark on this definition: it is Dedekind's definition; "finite" was never defined, and the agreement between "infinite" in this sense and "not finite" uses (a weak form of) the axiom of choice.
\begin{remark}
	Here a set is \emph{finite} if it is empty or equipotent to $\{1,2,\dots,n\}$ for some $n\in\Z^+$. The definition above is \emph{Dedekind's} definition of an infinite set. The pigeonhole argument shows that a Dedekind-infinite set is not finite; the converse (every set that is not finite is Dedekind-infinite) is also true, but its proof uses a weak form of the axiom of choice.
\end{remark}

\begin{example}
	Let $A=\{0,1,2,\dots\}$ and $B=\{1,2,3,\dots\}$. Show that $A\approx B$ and that $A$ is infinite.
	
	\vspace{0.5em}
	\noindent \textbf{Solution:}
	Define: 
	\[
	f:A\to B, \qquad f(n)=n+1.
	\]
	
	\begin{itemize}
		\item \textbf{$f$ is injective:} if $f(n)=f(m)$, then $n+1=m+1$, so $n=m$.
		\item \textbf{$f$ is surjective:} if $b\in B$, then $b-1\in A$ and $f(b-1)=b$.
	\end{itemize}
	
	\noindent Hence $f$ is a bijection, so $A\approx B$. Since $B\subset A$, the set $A$ is equipotent to one of its  proper subset. Therefore $A$ is infinite.
\end{example}

\begin{example}
	Show that $\mathbb{Z}^+\approx 2\mathbb{Z}^+$.
	
	\vspace{0.5em}
	\noindent \textbf{Solution:}
	Define:
	\[
	f:\mathbb{Z}^+\to 2\mathbb{Z}^+, \qquad f(n)=2n.
	\] 
	
	\begin{itemize}
		\item \textbf{$f$ is injective:} if $f(n)=f(m)$, then $2n=2m$, so $n=m$.
		\item \textbf{$f$ is surjective:} if $y\in 2\mathbb{Z}^+$, then $y=2n$ for some $n\in\mathbb{Z}^+$, so $f(n)=y$.
	\end{itemize}
	
	\noindent Hence $f$ is a bijection, and therefore $\mathbb{Z}^+\approx 2\mathbb{Z}^+$.
\end{example}

% Muhammed Alaa El-Din: I made the meaning of 2Z^+ + 1 explicit ({3, 5, 7, ...}, so 1 is not included), which is what makes f(n) = n + 1 onto.
\begin{example}
	Show that $2\mathbb{Z}^+\approx 2\mathbb{Z}^++1$, where $2\Z^+=\{2,4,6,\dots\}$ and $2\Z^++1=\{3,5,7,\dots\}$.
	
	\vspace{0.5em}
	\noindent \textbf{Solution:}
	Define:
	\[
	f:2\mathbb{Z}^+\to 2\mathbb{Z}^++1, \qquad f(n)=n+1.
	\]  If $n$ is even, then $n+1$ is odd, so $f$ is well defined.
	
	\begin{itemize}
		\item \textbf{$f$ is injective:} if $f(n)=f(m)$, then $n+1=m+1$, so $n=m$.
		\item \textbf{$f$ is surjective:} if $k\in 2\mathbb{Z}^++1$, then $k\ge3$, so $k-1\in 2\mathbb{Z}^+$ and $f(k-1)=k$.
	\end{itemize}
	
	\noindent Hence $f$ is a bijection, and therefore $2\mathbb{Z}^+\approx 2\mathbb{Z}^++1$.
\end{example}	

\subsection{Equipotent Intervals in $\R$}

\begin{example}
	\label{exp:openinteq}
	Let $a,b\in\R$ with $a<b$. Show that $\oi{0}{1}\approx\oi{a}{b}$.
	
	\vspace{0.5em}
	\noindent \textbf{Solution:}
	Define:
	\[
	f:\oi{0}{1}\to\oi{a}{b}, \qquad f(x)=a+(b-a)x .
	\]

	If $0<x<1$, then $a<a+(b-a)x<b$ because $b-a>0$, so $f$ is well defined.
	
	\begin{itemize}
		\item \textbf{$f$ is injective:} if $f(x)=f(x')$, then $a+(b-a)x=a+(b-a)x'$, so $(b-a)x=(b-a)x'$. Since $b-a\neq 0$, we get $x=x'$.
		\item \textbf{$f$ is surjective:} if $y\in\oi{a}{b}$, put $x=\dfrac{y-a}{b-a}$. Then $0<x<1$ and $f(x)=a+(y-a)=y$.
	\end{itemize}
	
	\noindent Hence $f$ is a bijection, and therefore $\oi{0}{1}\approx\oi{a}{b}$.
	
\end{example}

\begin{corollary}
	Any two bounded open intervals are equipotent:
	\[
	\oi{a}{b}\approx\oi{c}{d}\qquad (a<b,\ c<d).
	\]
\end{corollary}

% Muhammed Alaa El-Din: I made symmetry explicit: the example gives ]0,1[ ~ ]a,b[, and symmetry is needed to turn it around.
\begin{proof}
	By example~\ref{exp:openinteq} and symmetry, $\oi{a}{b}\approx\oi{0}{1}$; by the same example, $\oi{0}{1}\approx\oi{c}{d}$. Conclude by transitivity.
\end{proof}

\noindent In fact, unbounded intervals are also equipotent to bounded ones.

\begin{example}
	Show that $\oi{0}{1}\approx\oi{0}{\infty}$.
	
	\vspace{0.5em}
	\noindent \textbf{Solution:}
	Define:
	\[
	f:\oi{0}{1}\to\oi{0}{\infty}, \qquad f(x)=\frac1x-1 .
	\]
	If $0<x<1$, then $\frac1x>1$, so $f(x)>0$ and $f$ is well defined.
	
	\begin{itemize}
		\item \textbf{$f$ is injective:} if $f(x)=f(x')$, then $\frac1x-1=\frac1{x'}-1$, so $\frac1x=\frac1{x'}$ and $x=x'$.
		\item \textbf{$f$ is surjective:} if $y\in\oi{0}{\infty}$, put $x=\dfrac{1}{y+1}$. Then $0<x<1$ and $f(x)=(y+1)-1=y$.
	\end{itemize}
	
	\noindent Hence $f$ is a bijection with inverse $y\mapsto\dfrac{1}{y+1}$, and therefore $\oi{0}{1}\approx\oi{0}{\infty}$.
\end{example}

\begin{example}
	Show that $\oi{-1}{1}\approx\oi{-\infty}{\infty}$.
	
	\vspace{0.5em}
	\noindent \textbf{Solution:}
	Define:
	\[
	f:\oi{-1}{1}\to\oi{-\infty}{\infty}, \qquad f(x)=\tan\!\left(\frac{\pi}{2}x\right).
	\]
	If $-1<x<1$, then $-\frac{\pi}{2}<\frac{\pi}{2}x<\frac{\pi}{2}$, so $f$ is well defined.
	
	\begin{itemize}
		\item \textbf{$f$ is injective:} $\tan$ is strictly increasing on $\left(-\frac{\pi}{2},\frac{\pi}{2}\right)$, so $f(x)=f(x')$ implies $\frac{\pi}{2}x=\frac{\pi}{2}x'$, hence $x=x'$.
		\item \textbf{$f$ is surjective:} if $y\in\R$, put $x=\dfrac{2}{\pi}\arctan y$. Then $-1<x<1$ and $f(x)=\tan(\arctan y)=y$.
	\end{itemize}
	
	\noindent Hence $f$ is a bijection, and therefore $\oi{-1}{1}\approx\oi{-\infty}{\infty}$.
\end{example}

\begin{example}
	Show that $f:\oi{-\infty}{\infty}\to\oi{-1}{1}$, $f(y)=\dfrac{y}{\sqrt{1+y^2}}$, is a bijection.
	
	\vspace{0.5em}
	\noindent \textbf{Solution:}
	Since $|y|<\sqrt{1+y^2}$ for every $y\in\R$, we have $|f(y)|<1$, so $f$ is well defined.
	
	\begin{itemize}
		\item \textbf{$f$ is injective:} suppose $f(y_1)=f(y_2)$, i.e.
		\[
		\frac{y_1}{\sqrt{1+y_1^2}}=\frac{y_2}{\sqrt{1+y_2^2}} .
		\]
		Then $y_1,y_2$ have the same sign (or both are $0$, and then $y_1=y_2$). Squaring,
		\[
		\frac{y_1^2}{1+y_1^2}=\frac{y_2^2}{1+y_2^2}
		\ \Longrightarrow\ y_1^2(1+y_2^2)=y_2^2(1+y_1^2)
		\ \Longrightarrow\ y_1^2=y_2^2 ,
		\]
		so $|y_1|=|y_2|$. Since they have the same sign, $y_1=y_2$.
		\item \textbf{$f$ is surjective:} if $t\in\oi{-1}{1}$, put $y=\dfrac{t}{\sqrt{1-t^2}}$. Then $1+y^2=\dfrac{1}{1-t^2}$, so
		\[
		f(y)=y\sqrt{1-t^2}=t .
		\]
	\end{itemize}
	
	\noindent Hence $f$ is a bijection with inverse $t\mapsto\dfrac{t}{\sqrt{1-t^2}}$, and therefore $\oi{-\infty}{\infty}\approx\oi{-1}{1}$.
\end{example}



% Muhammed Alaa El-Din: I made the remaining cases explicit: the examples only treat ]0,\infty[ and R, and the other unbounded open intervals follow by translations and reflections.
\begin{takeaway}
	Any two non-trivial open intervals in $\R$ (bounded or not) are equipotent. The remaining unbounded cases follow from the examples by translations and reflections: $x\mapsto x+a$ gives $\oi{0}{\infty}\approx\oi{a}{\infty}$, and $x\mapsto b-x$ gives $\oi{0}{\infty}\approx\oi{-\infty}{b}$.
\end{takeaway}

\subsection{Open Versus Half-Open and Closed Intervals}

\begin{example}
	$\oi{0}{1}\approx\left]0,1\right]$. Define $f:\left]0,1\right]\to\oi{0}{1}$ by
	\[
	f(x)=\begin{cases}
		x & x\neq\dfrac1n\ \ (n\in\Z^+),\\[2mm]
		\dfrac1{n+1} & x=\dfrac1n\ \ (n\in\Z^+).
	\end{cases}
	\]
	The idea is to shift the points $1,\tfrac12,\tfrac13,\dots$ one step towards $0$ and leave all other points fixed:
	\begin{center}

\begin{corollary}
	$\oi{a}{b}\approx\left]a,b\right]$ for $a<b$.
\end{corollary}

\begin{proof}
	The affine bijection $x\mapsto a+(b-a)x$ maps $\left]0,1\right]$ onto $\left]a,b\right]$ and $\oi{0}{1}$ onto $\oi{a}{b}$. Combine with the example and use transitivity.
\end{proof}

\begin{remark}
	Let $a,b\in\R$ with $a<b$. Then $\left]a,b\right]\approx\left[a,b\right[$ by the ``mirror image'' of the interval about its midpoint $\dfrac{a+b}{2}$.
	
	\vspace{0.5em}
	\noindent \textbf{Details:}
	Define:
	\[
	f:\left]a,b\right]\to\left[a,b\right[, \qquad f(x)=a+b-x .
	\]
	
	\noindent \textbf{$f$ is well defined:} let $x\in\left]a,b\right]$, so $a<x\leq b$. Multiplying by $-1$ reverses the inequalities:
	\[
	-b\leq -x<-a .
	\]
	Adding $a+b$ to every part gives
	\[
	a\leq a+b-x<b ,
	\]
	that is, $f(x)\in\left[a,b\right[$.
	
	\begin{itemize}
		\item \textbf{$f$ is injective:} if $f(x)=f(x')$, then $a+b-x=a+b-x'$, so $x=x'$.
		\item \textbf{$f$ is surjective:} let $y\in\left[a,b\right[$, so $a\leq y<b$. Put $x=a+b-y$. Multiplying by $-1$ and adding $a+b$ as above gives $a<a+b-y\leq b$, that is, $x\in\left]a,b\right]$. Moreover
		\[
		f(x)=a+b-(a+b-y)=y .
		\]
	\end{itemize}
	
	\noindent Hence $f$ is a bijection, and therefore $\left]a,b\right]\approx\left[a,b\right[$.
	
\end{remark}

\begin{takeaway}
	Any two non-trivial intervals in $\R$ (i.e.\ intervals with more than one point), whether open, half-open or closed, bounded or not, are equipotent; the cases not written out above are handled in the same way. In particular, if $-\infty\le a<b\le\infty$ and $-\infty\le c<d\le\infty$, then

	\]
\end{takeaway}

\subsection{Countable Sets}

% Muhammed Alaa El-Din: I made a note that "subset of Z" can equivalently be "subset of Z^+" (the more common definition), because Z ~ Z^+ (proved at the end of this lecture).
\begin{definition}{Countable Set}
	A set $A$ is called \textbf{countable} if it is equipotent to a subset of $\Z$. Infinite countable sets are called \textbf{denumerable}. (Since $\Z\approx\Z^+$, see below, one may equivalently use subsets of $\Z^+$.)
\end{definition}


\begin{remark}
	An infinite set is denumerable if and only if it can be listed in a sequence of distinct terms.
\end{remark}

\begin{theorem}
	$\oi{0}{1}$ is not denumerable.
\end{theorem}

% Muhammed Alaa El-Din: I made the choice of digits safe. With digits 0 and 6, y could be 0.000... = 0 (if every diagonal digit is >= 5), and 0 is not in ]0,1[, so there would be no contradiction. Using digits 4 and 6 guarantees 0 < y < 1 and that y has a unique decimal expansion. I also fixed the expansions of the x_n (those not ending in 9's) and corrected "the converse" to "the contrary".
\begin{proof}
	Suppose the contrary. That is, $\oi{0}{1}=\{x_1,x_2,x_3,\dots\}$. Write each $x_k$ in its decimal expansion, choosing the expansion that does not end in an infinite string of $9$'s:
	\begin{align*}
		x_1 &= 0.\,x_1^{(1)}\,x_1^{(2)}\cdots x_1^{(n)}\cdots\\
		x_2 &= 0.\,x_2^{(1)}\,x_2^{(2)}\cdots x_2^{(n)}\cdots\\
		&\ \,\vdots\\
		x_n &= 0.\,x_n^{(1)}\,x_n^{(2)}\cdots x_n^{(n)}\cdots
	\end{align*}
	
	\noindent Define $y=0.\,y^{(1)}\,y^{(2)}\cdots y^{(n)}\cdots$ where
	\[
	y^{(n)}=\begin{cases}
		4 & \text{if } x_n^{(n)}\geq 5,\\[1mm]
		6 & \text{if } x_n^{(n)}< 5 .
	\end{cases}
	\]
	Since every digit of $y$ is $4$ or $6$, we have $0.4\le y\le0.7$, so $y\in\oi{0}{1}$; moreover $y$ contains neither $0$'s nor $9$'s, so it has only one decimal expansion. Now $y^{(n)}\neq x_n^{(n)}$ for every $n$, so the expansion of $y$ differs from the chosen expansion of $x_n$ in the $n$-th digit; since $y$ has a unique expansion, $y\neq x_n$ $(\forall\, n\in\Z^+)$. This is a contradiction, since $y\in\oi{0}{1}$ but $y$ is not in the list $\{x_1,x_2,x_3,\dots\}$.
\end{proof}

\begin{theorem}
	$\Z^+\times\Z^+\approx\Z^+$.
\end{theorem}



% Muhammed Alaa El-Din: I made the "onto" part justified (repeatedly divide by 2) instead of only asserting it.
\begin{proof}
	Define:
	\[
	f:\Z^+\times\Z^+\to\Z^+, \qquad f(m,n)=2^{m-1}(2n-1).
	\]
	\textbf{$f$ is onto:} every positive integer $k$ can be written as $2^{m-1}(2n-1)$, namely a power of $2$ times an odd number: divide $k$ by $2$ as many times as possible, say $m-1\ge0$ times; the quotient is odd, so it equals $2n-1$ for some $n\in\Z^+$.
	
	\smallskip
	\noindent\textbf{$f$ is one-to-one:} suppose $2^{m_1-1}(2n_1-1)=2^{m_2-1}(2n_2-1)$ and, say, $m_1>m_2$. Dividing by $2^{m_2-1}$ gives
	\[
	2^{m_1-m_2}(2n_1-1)=2n_2-1,
	\]
	where the left side is even and the right side is odd, a contradiction. So $m_1=m_2$, and then $2n_1-1=2n_2-1$, hence $n_1=n_2$.
	
	
\end{proof}

\begin{example}[$\Q$ is countable]
	Let
	\[
	S=\bigl\{(m,n): m\in\Z,\ n\in\Z^+,\ \gcd(m,n)=1\bigr\}, \qquad f:S\to\Q, \quad f(m,n)=\frac mn .
	\]
	Every rational number has a unique representation $\tfrac mn$ in lowest terms with $n>0$ (for $0$ it is $\tfrac01$), so $f$ is a bijection and
	\[
	\Q\approx S,
	\]
	a subset of $\Z\times\Z^+\approx\Z^+$. Hence $\Q$ is countable (by the exercise on subsets of countable sets); since $\Q$ is infinite, it is denumerable.
\end{example}
